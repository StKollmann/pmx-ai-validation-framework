[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; Define the absorption compartment (cmt=1)
; This compartment receives the dose via zero-order absorption
compartment(cmt=1, amount=Aabs)

; Define the central compartment (cmt=2)
compartment(cmt=2, amount=Ac, volume=V, concentration=Cc)

; Define the peripheral compartment (cmt=3)
compartment(cmt=3, amount=A2, volume=V2)

; Zero-order absorption into the absorption compartment (cmt=1)
absorption(cmt=1, Tk0)

; First-order transfer from the absorption compartment (cmt=1) to the central compartment (cmt=2)
; This represents the "first-order absorption" into the central compartment, sequential to the zero-order phase
transfer(from=1, to=2, kt=ka)

; Transfer between central (cmt=2) and peripheral (cmt=3) compartments
transfer(from=2, to=3, kt=Q/V)
transfer(from=3, to=2, kt=Q/V2)

; Linear elimination from the central compartment (cmt=2)
elimination(cmt=2, Cl)

OUTPUT:
output = {Cc}
