[LONGITUDINAL]
DESCRIPTION:
Exposure-response model with Cavg as exposure, an effect compartment, and Emax response.

input = {E0, ke0, Emax, EC50, Cavg}
Cavg = {use=regressor}

EQUATION:
; Initial condition for the effect compartment concentration
Ce_0 = 0

; Differential equation for the effect compartment
ddt_Ce = ke0 * (Cavg - Ce)

; Emax pharmacodynamic model
E = E0 + (Emax * Ce) / (EC50 + Ce)

OUTPUT:
output = {E}
