[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = kdeg*Circ0/ktr
Circ_0 = Circ0

Cc = Ac/V
CN = Circ
Inhib = Cc/(EC50 + Cc)
Feedback = (Circ0/Circ)^beta
Prod = kprod*Prol*(1 - Inhib)*Feedback

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = Prod - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

OUTPUT:
output = {Cc, CN}
