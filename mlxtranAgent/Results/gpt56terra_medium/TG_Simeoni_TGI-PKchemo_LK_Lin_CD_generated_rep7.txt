[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau, ka_cis, k_cis, Vd_cis, ka_pem, k_pem, Vd_pem}

PK:
compartment(cmt=2, amount=Acis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

compartment(cmt=3, amount=Apem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff
psi = 20
EXPOSURE = C_cis + C_pem
killRate = kkill*EXPOSURE
growthRate = kge*TS/(1 + (kge*TS/kgl)^psi)^(1/psi)
TS_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0
ddt_TS = growthRate - killRate*TS
ddt_TS1 = killRate*TS - TS1/tau
ddt_TS2 = TS1/tau - TS2/tau
ddt_TS3 = TS2/tau - TS3/tau
TotalTS = TS + TS1 + TS2 + TS3

OUTPUT:
output = {TotalTS}
