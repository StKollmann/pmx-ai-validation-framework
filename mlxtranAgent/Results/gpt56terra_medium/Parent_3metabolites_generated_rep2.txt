[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
depot(target=Ad1, p=0.1284)
depot(target=Ad2, p=0.5952)
depot(target=Ad3, p=0.2381)
depot(target=Ad4, p=0.0383)

EQUATION:
odeType = stiff

ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

ddt_Ap = (Q/V1)*Ac - (Q/V2)*Ap
ddt_Ac = ka1*Ad1 - (Cl/V1)*Ac - (Q/V1)*Ac + (Q/V2)*Ap - kpm2*Ac - kpm3*Ac - kpm4*Ac

ddt_Am2 = ka2*Ad2 + kpm2*Ac - (Clm2/V1)*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ac - (Clm3/V1)*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ac - (Clm4/V1)*Am4

Cp = Ac/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
