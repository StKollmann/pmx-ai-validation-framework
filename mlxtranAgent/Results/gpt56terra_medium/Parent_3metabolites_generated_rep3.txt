[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383
depot(target=Adp, adm=1, p=F1)
depot(target=Adm2, adm=1, p=F2)
depot(target=Adm3, adm=1, p=F3)
depot(target=Adm4, adm=1, p=F4)

EQUATION:
odeType = stiff
Cp = Ap/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

ddt_Adp = -ka1*Adp
ddt_Adm2 = -ka2*Adm2
ddt_Adm3 = -ka3*Adm3
ddt_Adm4 = -ka4*Adm4
ddt_Ap = ka1*Adp - (Cl/V1)*Ap - (Q/V1)*Ap + (Q/V2)*App - kpm2*Ap - kpm3*Ap - kpm4*Ap
ddt_App = (Q/V1)*Ap - (Q/V2)*App
ddt_Am2 = ka2*Adm2 + kpm2*Ap - (Clm2/V1)*Am2
ddt_Am3 = ka3*Adm3 + kpm3*Ap - (Clm3/V1)*Am3
ddt_Am4 = ka4*Adm4 + kpm4*Ap - (Clm4/V1)*Am4

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
