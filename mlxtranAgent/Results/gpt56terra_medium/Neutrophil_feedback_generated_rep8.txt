[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff
Cc = Ac/V
Inhib = Cc/(EC50 + Cc)
Prolif_0 = kdeg*Circ0/ktr
CN_0 = Circ0
ddt_Ac = -(Cl/V)*Ac
ddt_Prolif = kprod*(1-Inhib)*(Circ0/CN)^beta*Prolif - ktr*Prolif
ddt_CN = ktr*Prolif - kdeg*CN

OUTPUT:
output = {Cc, CN}
