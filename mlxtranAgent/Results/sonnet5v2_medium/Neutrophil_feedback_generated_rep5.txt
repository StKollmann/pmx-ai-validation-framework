[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ad, ka)

EQUATION:
odeType = stiff

; --- PK ---
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; --- PD : turnover model with feedback ---
Prol_0 = Circ0*kdeg/ktr
Circ_0 = Circ0

Edrug = Cc/(Cc + EC50)
Feedback = (Circ0/Circ)^beta

ddt_Prol = kprod*(1 - Edrug)*Feedback*Prol - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
