[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate derived from R0
beta = R0*delta*c/(p*T0)

; initial conditions
T_0  = T0
I1_0 = 0
I2_0 = 0
V_0  = 10^V0

; ODEs: target cell limited model with eclipse phase
; loss of virion due to infection included in ddt_V (-beta*T*V term)
ddt_T  = -beta*T*V
ddt_I1 =  beta*T*V - kI*I1
ddt_I2 =  kI*I1 - delta*I2
ddt_V  =  p*I2 - c*V - beta*T*V

; log10 viral load with lower saturation at -5
logV = log10(V)
if logV < -5
    LV = -5
else
    LV = logV
end

OUTPUT:
output = {LV}
