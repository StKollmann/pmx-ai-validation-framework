[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

depot(target=Ad1, adm=1, p=F1)
depot(target=Ad2, adm=1, p=F2)
depot(target=Ad3, adm=1, p=F3)
depot(target=Ad4, adm=1, p=F4)

EQUATION:
odeType = stiff

; Absorption depots (fraction of dose feeding parent / each metabolite pathway)
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; Parent: 2-compartment model, with additional clearance to metabolites (first-pass/systemic conversion)
ddt_Ac  = ka1*Ad1 - (Cl+Clm2+Clm3+Clm4)/V1*Ac - Q/V1*Ac + Q/V2*Ap2
ddt_Ap2 = Q/V1*Ac - Q/V2*Ap2

; Metabolite 2: direct first-pass fraction (ka2*Ad2) + conversion from parent (Clm2), eliminated at kpm2
ddt_Am2 = ka2*Ad2 + Clm2/V1*Ac - kpm2*Am2

; Metabolite 3
ddt_Am3 = ka3*Ad3 + Clm3/V1*Ac - kpm3*Am3

; Metabolite 4
ddt_Am4 = ka4*Ad4 + Clm4/V1*Ac - kpm4*Am4

Cp  = Ac/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
