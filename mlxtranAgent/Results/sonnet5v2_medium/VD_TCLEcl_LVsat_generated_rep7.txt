[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; derived infection rate from basic reproduction number
beta = R0*delta*c/(p*T0)

; initial conditions
T_0  = T0
I1_0 = 0
I2_0 = 0
V_0  = 10^V0

; ODE system: target cell limited model with eclipse phase
; virions are lost upon infecting a target cell (beta*V*T term in ddt_V)
ddt_T  = -beta*V*T
ddt_I1 = beta*V*T - kI*I1
ddt_I2 = kI*I1 - delta*I2
ddt_V  = p*I2 - c*V - beta*V*T

; log10 viral load with lower saturation at -5
logV = log10(V)
if logV < -5
  LV = -5
else
  LV = logV
end

OUTPUT:
output = {LV}
