[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate derived from basic reproduction number
beta = R0*c*delta/(p*T0)

; initial conditions
T_0  = T0
I1_0 = 0
I2_0 = 0
V_0  = 10^V0

; ODE system: Target - Eclipse - Infected - Virus
ddt_T  = -beta*V*T
ddt_I1 =  beta*V*T - kI*I1
ddt_I2 =  kI*I1 - delta*I2
ddt_V  =  p*I2 - c*V - beta*V*T

; log10 viral load with lower saturation at -5
LV = max(log10(V), -5)

OUTPUT:
output = {LV}
