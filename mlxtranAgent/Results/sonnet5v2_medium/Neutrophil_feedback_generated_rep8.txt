[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ad, ka)

EQUATION:
odeType = stiff

; PK
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; PD initial conditions (steady-state assumption)
Prol_0 = Circ0
Circ_0 = Circ0

; Drug inhibition of proliferation (Emax model, Emax=1)
Edrug = Cc/(Cc + EC50)

; Feedback from circulating compartment on proliferation
feedback = (Circ0/Circ)^beta

; Turnover PD model
ddt_Prol = kprod*Prol*(1-Edrug)*feedback - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
