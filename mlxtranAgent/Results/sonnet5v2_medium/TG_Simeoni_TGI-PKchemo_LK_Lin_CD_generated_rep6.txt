[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; --- Cisplatin PK (fixed parameters) ---
ka_cis = 66.5421
k_cis = 0.2868
Vd_cis = 65.1131
compartment(cmt=2, amount=Ac_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; --- Pemetrexed PK (fixed parameters) ---
ka_pem = 28.6
k_pem = 2.1328
Vd_pem = 102.7673
compartment(cmt=3, amount=Ac_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; psi fixed to 20 (switch between exponential and linear growth)
psi = 20

; total drug exposure driving the tumor kill
EXPOSURE = C_cis + C_pem

; initial conditions
TSprolif_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

; Simeoni unperturbed growth (exponential-linear switch) and log-kill effect
g = kge*TSprolif / (1+(kge/kgl*TSprolif)^psi)^(1/psi)
kill = kkill*EXPOSURE*TSprolif

ddt_TSprolif = g - kill
ddt_TS1 = kill - TS1/tau
ddt_TS2 = TS1/tau - TS2/tau
ddt_TS3 = TS2/tau - TS3/tau

TotalTS = TSprolif + TS1 + TS2 + TS3

OUTPUT:
output = {TotalTS}
