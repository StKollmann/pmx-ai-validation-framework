[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}

ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
{Cc} = pkmodel(ka, V, Cl)

EQUATION:
; --- Baseline steady-state, correlated with R0 (PD not at steady-state at t=0) ---
Rss = intercept + slope*R0
kin = kout*Rss

; --- Turnover model with stimulation of production by the drug ---
R_0 = R0
ddt_R = kin*(1 + Emax*Cc/(EC50+Cc)) - kout*R

OUTPUT:
output = {Cc, R}
