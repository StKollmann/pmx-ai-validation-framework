[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; --- derived infection rate from R0 ---
beta = R0*delta*c/(p*T0)

; --- initial conditions ---
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; --- ODE system: target cell limited model with eclipse phase ---
ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

; --- log10 viral load with lower saturation at -5 ---
if V > 1e-5
    LV = log10(V)
else
    LV = -5
end

OUTPUT:
output = {LV}
