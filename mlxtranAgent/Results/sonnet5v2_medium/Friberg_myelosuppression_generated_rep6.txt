[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
depot(target=Ad, ka)

EQUATION:
; --- PK: one-compartment with first-order absorption and linear elimination ---
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; --- PD: Friberg semi-mechanistic myelosuppression model ---
ktr    = 4/MTT
kprol  = ktr
kcirc  = ktr

; Drug inhibition of proliferation (Emax model)
Edrug = Emax*Cc/(EC50 + Cc)

; Initial conditions: all PD compartments start at baseline Circ0
Prol_0     = Circ0
Transit1_0 = Circ0
Transit2_0 = Circ0
Transit3_0 = Circ0
Circ_0     = Circ0

; Homeostatic feedback on proliferation
Feedback = (Circ0/Circ)^gam

ddt_Prol     = kprol*Prol*(1 - Edrug)*Feedback - ktr*Prol
ddt_Transit1 = ktr*Prol     - ktr*Transit1
ddt_Transit2 = ktr*Transit1 - ktr*Transit2
ddt_Transit3 = ktr*Transit2 - ktr*Transit3
ddt_Circ     = ktr*Transit3 - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
