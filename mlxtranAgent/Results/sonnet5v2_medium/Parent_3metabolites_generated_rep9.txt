[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; fixed dose fractions (first-pass split)
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

depot(target=Ad1, p=F1, adm=1)
depot(target=Ad2, p=F2, adm=1)
depot(target=Ad3, p=F3, adm=1)
depot(target=Ad4, p=F4, adm=1)

EQUATION:
odeType=stiff

; volumes of metabolite central compartments fixed to parent volume V1
Vm2 = V1
Vm3 = V1
Vm4 = V1

; rate constants for parent 2-compartment model
k12 = Q/V1
k21 = Q/V2
ke  = Cl/V1

; absorption depots (dose fractions)
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; parent drug - central and peripheral compartments
ddt_Ac = ka1*Ad1 - ke*Ac - k12*Ac + k21*Ap - (kpm2 + kpm3 + kpm4)*Ac
ddt_Ap = k12*Ac - k21*Ap

; metabolites - one compartment each, fed by direct first-pass absorption and by parent transformation
ddt_Am2 = ka2*Ad2 + kpm2*Ac - (Clm2/Vm2)*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ac - (Clm3/Vm3)*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ac - (Clm4/Vm4)*Am4

Cp  = Ac/V1
Cm2 = Am2/Vm2
Cm3 = Am3/Vm3
Cm4 = Am4/Vm4

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
