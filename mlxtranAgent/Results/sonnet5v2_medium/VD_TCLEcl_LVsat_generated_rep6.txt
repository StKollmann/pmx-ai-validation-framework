[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate derived from basic reproduction number
beta = R0*c*delta/(p*T0)

; initial conditions
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; target cell limited model with eclipse phase
; virions are lost both by infecting cells (beta*T*V) and by clearance (c*V)
ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

; log10 viral load with lower saturation at -5
LV = max(log10(V), -5)

OUTPUT:
output = {LV}
