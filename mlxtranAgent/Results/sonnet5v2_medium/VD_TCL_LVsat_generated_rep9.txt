[LONGITUDINAL]
input = {beta, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

t_0 = 0
T_0 = T0
I_0 = 0
V_0 = V0

ddt_T = -beta*T*V
ddt_I =  beta*T*V - delta*I
ddt_V =  p*I - c*V

; log10 viral load with lower saturation at -5
logV = log10(V)
if logV < -5
    LV = -5
else
    LV = logV
end

OUTPUT:
output = {LV}
