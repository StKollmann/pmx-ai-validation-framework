[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; Fixed dose fractions (first-pass effect split)
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

; Doses split into 4 depots according to fixed fractions
depot(target=Ad1, p=F1)
depot(target=Ad2, p=F2)
depot(target=Ad3, p=F3)
depot(target=Ad4, p=F4)

EQUATION:
odeType=stiff

; metabolite central volumes fixed to parent central volume
Vm2 = V1
Vm3 = V1
Vm4 = V1

; absorption depots
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; parent 2-compartment model with conversion to metabolites
ddt_Ac1 = ka1*Ad1 - (Cl/V1)*Ac1 - (Q/V1)*Ac1 + (Q/V2)*Ac2 - kpm2*Ac1 - kpm3*Ac1 - kpm4*Ac1
ddt_Ac2 = (Q/V1)*Ac1 - (Q/V2)*Ac2

; metabolites (one compartment each): formed via first-pass (ka2/3/4) and via parent conversion (kpm2/3/4)
ddt_Acm2 = ka2*Ad2 + kpm2*Ac1 - (Clm2/Vm2)*Acm2
ddt_Acm3 = ka3*Ad3 + kpm3*Ac1 - (Clm3/Vm3)*Acm3
ddt_Acm4 = ka4*Ad4 + kpm4*Ac1 - (Clm4/Vm4)*Acm4

Cp = Ac1/V1
Cm2 = Acm2/Vm2
Cm3 = Acm3/Vm3
Cm4 = Acm4/Vm4

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
