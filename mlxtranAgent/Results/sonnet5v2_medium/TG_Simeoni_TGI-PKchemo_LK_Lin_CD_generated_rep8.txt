[LONGITUDINAL]
input = {TS0, kge, kgl, psi, kkill, tau, ka_cis, k_cis, Vd_cis, ka_pem, k_pem, Vd_pem}

PK:
; Cisplatin PK (cmt=2, adm=2)
compartment(cmt=2, amount=Ac_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; Pemetrexed PK (cmt=3, adm=1)
compartment(cmt=3, amount=Ac_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; total drug exposure
EXPOSURE = C_cis + C_pem

; initial condition for the proliferating tumor compartment
TSprol_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

; unperturbed (Simeoni) growth function: exponential-to-linear switch governed by psi
growth = kge*TSprol / (1 + (kge/kgl*TSprol)^psi)^(1/psi)

; log-kill drug effect acting on proliferating compartment, linear in exposure
kill = kkill*EXPOSURE*TSprol

ddt_TSprol = growth - kill
ddt_TS1 = kill - TS1/tau
ddt_TS2 = TS1/tau - TS2/tau
ddt_TS3 = TS2/tau - TS3/tau

TotalTS = TSprol + TS1 + TS2 + TS3

OUTPUT:
output = {TotalTS}
