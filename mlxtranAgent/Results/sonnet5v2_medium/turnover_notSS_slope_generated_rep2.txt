[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
Cc = pkmodel(ka, V, Cl)

EQUATION:
; Derive Rss (steady-state response) from R0 via linear relationship R0 = intercept + slope*Rss
Rss = (R0 - intercept)/slope
kin = Rss*kout

; Initial condition: response not at steady-state at t=0
R_0 = R0

; Turnover model with stimulation of production by drug
ddt_R = kin*(1 + Emax*Cc/(EC50+Cc)) - kout*R

OUTPUT:
output = {Cc, R}
