[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; secondary parameter: infection rate derived from R0
beta = R0*c*delta/(p*T0)

; initial conditions
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; ODE system: target cell limited model with eclipse phase
; virions are lost upon infecting a target cell (-beta*V*T term in ddt_V)
ddt_T = -beta*V*T
ddt_E =  beta*V*T - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*V*T

; log10 viral load with lower saturation at -5
LV = max(log10(V), -5)

OUTPUT:
output = {LV}
