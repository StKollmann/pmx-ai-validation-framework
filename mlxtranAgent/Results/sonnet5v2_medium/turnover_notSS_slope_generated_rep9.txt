[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
{Cc} = pkmodel(ka, V, Cl)

EQUATION:
; Steady-state response (Rss) is linearly correlated with the baseline response R0
Rss = intercept + slope*R0
; Production rate constant derived from Rss and kout (turnover steady-state without drug)
kin = kout*Rss

; Initial condition: PD is NOT at steady-state at t=0, so R(0) = R0 (not Rss)
R_0 = R0

; Turnover model with stimulation of production by the drug
ddt_R = kin*(1 + Emax*Cc/(EC50+Cc)) - kout*R

OUTPUT:
output = {Cc, R}
