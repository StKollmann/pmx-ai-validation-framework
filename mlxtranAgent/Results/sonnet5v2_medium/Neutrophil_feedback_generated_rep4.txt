[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ad, ka)

EQUATION:
odeType = stiff

; PK
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; Initial conditions for PD compartments (system at steady-state at baseline)
Prol_0 = kprod/ktr
Circ_0 = Circ0

; Drug inhibition of proliferation (Emax model)
Edrug = Cc/(EC50+Cc)

; Feedback from circulating neutrophils on proliferation
Feedback = (Circ0/Circ)^beta

; PD turnover model: proliferating (Prol) and circulating (Circ) neutrophils
ddt_Prol = kprod*(1-Edrug)*Feedback - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
