[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; Infection rate derived from basic reproduction number
beta = R0*c*delta/(p*T0)

; Initial conditions
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; ODEs: target cells, eclipse phase, productively infected cells, virus
ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

; Log10 viral load with lower saturation at -5
LV = max(log10(V), -5)

OUTPUT:
output = {LV}
