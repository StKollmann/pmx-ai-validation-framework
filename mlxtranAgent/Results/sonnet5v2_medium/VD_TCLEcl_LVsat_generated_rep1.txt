[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate constant derived from R0
beta = R0*c*delta/(p*T0)

; initial conditions
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; ODE system: target cell limited model with eclipse phase
; virion loss due to infection included in ddt_V (term -beta*T*V)
ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

; log10 viral load with lower saturation at -5
Vsafe = max(V, 1e-10)
LV = max(log10(Vsafe), -5)

OUTPUT:
output = {LV}
