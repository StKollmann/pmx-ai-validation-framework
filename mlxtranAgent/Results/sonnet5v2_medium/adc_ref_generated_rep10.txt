[LONGITUDINAL]
input = {k12, k21, V, k10, V_payload, k_payload}

PK:
; ADC disposition: central + peripheral compartments, IV administration
compartment(cmt=1, amount=Ac_ADC, volume=V, concentration=Cc_ADC)
iv(cmt=1, adm=1)
elimination(cmt=1, k=k10)
peripheral(k12, k21)

EQUATION:
; Proteolytic release of payload from ADC in central compartment
; DAR (drug-to-antibody ratio) is constant = 4
DAR = 4
releaseRate = DAR * k10 * Ac_ADC

; Payload one-compartment disposition (ODE-based)
ddt_Ap = releaseRate - k_payload*Ap
Cc_payload = Ap/V_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
