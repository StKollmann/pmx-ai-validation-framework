[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; Zero-order input into absorption depot compartment Ad
depot(target=Ad, Tk0=Tk0)

EQUATION:
; First-order absorption from Ad to central compartment Ac (manual ODE)
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac - (Q/V)*Ac + (Q/V2)*Ap
ddt_Ap = (Q/V)*Ac - (Q/V2)*Ap

Cc = Ac/V

OUTPUT:
output = {Cc}
