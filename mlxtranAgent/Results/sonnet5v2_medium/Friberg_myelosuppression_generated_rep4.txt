[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
depot(target=Ad)

EQUATION:
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; --- Transit rate constant from mean transit time (3 transit cmts + circulating) ---
ktr = 4/MTT
kprol = ktr
kcirc = ktr

; --- Initial conditions: all PD compartments at baseline ---
Prol_0 = Circ0
Transit1_0 = Circ0
Transit2_0 = Circ0
Transit3_0 = Circ0
Circ_0 = Circ0

; --- Drug effect (Emax model) on proliferation ---
Edrug = Emax*Cc/(EC50 + Cc)

; --- Feedback from circulating compartment ---
Feedback = (Circ0/Circ)^gam

ddt_Prol = kprol*Prol*(1 - Edrug)*Feedback - ktr*Prol
ddt_Transit1 = ktr*Prol - ktr*Transit1
ddt_Transit2 = ktr*Transit1 - ktr*Transit2
ddt_Transit3 = ktr*Transit2 - ktr*Transit3
ddt_Circ = ktr*Transit3 - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
