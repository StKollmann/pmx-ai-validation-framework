[LONGITUDINAL]
input = {ka1, ka2, ka3, ka4, V1, Cl, Q, V2, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; Fixed first-pass fractions of the dose (0.1284+0.5952+0.2381+0.0383=1)
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

; Dose split into 4 depots: F1 -> parent absorption, F2/F3/F4 -> direct first-pass formation of M2/M3/M4
depot(target=Ad1,  p=F1, adm=1)
depot(target=Ad2m, p=F2, adm=1)
depot(target=Ad3m, p=F3, adm=1)
depot(target=Ad4m, p=F4, adm=1)

EQUATION:
odeType=stiff

; --- Absorption depots ---
ddt_Ad1  = -ka1*Ad1
ddt_Ad2m = -ka2*Ad2m
ddt_Ad3m = -ka3*Ad3m
ddt_Ad4m = -ka4*Ad4m

; --- Parent: 2-compartment disposition + systemic transformation to metabolites ---
ddt_Ac1 = ka1*Ad1 - (Cl/V1)*Ac1 - (Q/V1)*Ac1 + (Q/V2)*Ac2 - kpm2*Ac1 - kpm3*Ac1 - kpm4*Ac1
ddt_Ac2 = (Q/V1)*Ac1 - (Q/V2)*Ac2

; --- Metabolites: one compartment each, volume fixed to parent V1 ---
ddt_Am2 = ka2*Ad2m + kpm2*Ac1 - (Clm2/V1)*Am2
ddt_Am3 = ka3*Ad3m + kpm3*Ac1 - (Clm3/V1)*Am3
ddt_Am4 = ka4*Ad4m + kpm4*Ac1 - (Clm4/V1)*Am4

; --- Concentrations ---
Cp  = Ac1/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
