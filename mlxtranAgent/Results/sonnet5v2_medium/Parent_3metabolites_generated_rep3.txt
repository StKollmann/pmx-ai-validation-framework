[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; Fixed first-pass fractions of the dose
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

; Dose is split into 4 fractions:
; F1 -> parent absorption depot
; F2, F3, F4 -> direct (first-pass) formation of metabolites 2, 3, 4
depot(target=Ad1, p=F1, adm=1)
depot(target=Ad2, p=F2, adm=1)
depot(target=Ad3, p=F3, adm=1)
depot(target=Ad4, p=F4, adm=1)

EQUATION:
odeType=stiff

; --- Absorption depots (manual absorption, Pattern A) ---
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; --- Parent: 2-compartment disposition + conversion to metabolites ---
ddt_Ac1 = ka1*Ad1 - (Cl/V1)*Ac1 - (Q/V1)*Ac1 + (Q/V2)*Ac2 - kpm2*Ac1 - kpm3*Ac1 - kpm4*Ac1
ddt_Ac2 = (Q/V1)*Ac1 - (Q/V2)*Ac2

; --- Metabolite 2 (1-compartment, volume fixed to V1) ---
ddt_Am2 = ka2*Ad2 + kpm2*Ac1 - (Clm2/V1)*Am2

; --- Metabolite 3 (1-compartment, volume fixed to V1) ---
ddt_Am3 = ka3*Ad3 + kpm3*Ac1 - (Clm3/V1)*Am3

; --- Metabolite 4 (1-compartment, volume fixed to V1) ---
ddt_Am4 = ka4*Ad4 + kpm4*Ac1 - (Clm4/V1)*Am4

; --- Concentrations ---
Cp  = Ac1/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
