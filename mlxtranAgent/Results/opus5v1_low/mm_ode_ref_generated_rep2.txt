[LONGITUDINAL]
input = {V, k, Vm, Km}

PK:
; IV bolus doses enter the central compartment directly
depot(target=Ac)

EQUATION:
odeType = stiff
Ac_0 = 0

; define concentration before use
Cc = Ac/V

; central compartment amount: linear + Michaelis-Menten elimination
ddt_Ac = -k*Ac - Vm*Cc/(Km + Cc)

OUTPUT:
output = {Cc}
