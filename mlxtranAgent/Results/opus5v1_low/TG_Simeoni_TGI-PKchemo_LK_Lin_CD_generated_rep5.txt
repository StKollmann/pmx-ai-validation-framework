[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; --- Cisplatin PK (cmt=2, adm=2) ---
ka_cis = 66.5421
k_cis  = 0.2868
Vd_cis = 65.1131
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; --- Pemetrexed PK (cmt=3, adm=1) ---
ka_pem = 28.6
k_pem  = 2.1328
Vd_pem = 102.7673
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

psi = 20

; initial conditions
x1_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

EXPOSURE = C_cis + C_pem

; total tumor size
TotalTS = x1 + TS1 + TS2 + TS3

; Simeoni growth function
growth = 2*kge*x1 / (1 + (2*kge*TotalTS/kgl)^psi)^(1/psi)

; killing (log-kill)
kill = kkill*EXPOSURE*x1

k1 = 1/tau

ddt_x1  = growth - kill
ddt_TS1 = kill - k1*TS1
ddt_TS2 = k1*TS1 - k1*TS2
ddt_TS3 = k1*TS2 - k1*TS3

OUTPUT:
output = {TotalTS}
