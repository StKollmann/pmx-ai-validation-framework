DESCRIPTION: Simeoni TGI with cisplatin + pemetrexed exposure

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; ---- fixed PK parameters ----
ka_cis = 66.5421
k_cis  = 0.2868
Vd_cis = 65.1131
ka_pem = 28.6
k_pem  = 2.1328
Vd_pem = 102.7673

; ---- cisplatin (cmt=2, adm=2) ----
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; ---- pemetrexed (cmt=3, adm=1) ----
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff
psi = 20

TS_0  = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

EXPOSURE = C_cis + C_pem

growth = 2*kge*TS / (1 + (2*kge*TS/kgl)^psi)^(1/psi)
kill   = kkill*EXPOSURE*TS

ddt_TS  = growth - kill
ddt_TS1 = kill    - TS1/tau
ddt_TS2 = TS1/tau - TS2/tau
ddt_TS3 = TS2/tau - TS3/tau

TotalTS = TS + TS1 + TS2 + TS3

OUTPUT:
output = {TotalTS}
