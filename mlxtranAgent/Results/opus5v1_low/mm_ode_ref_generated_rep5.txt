[LONGITUDINAL]
input = {V, k, Vm, Km}

PK:
; IV bolus doses go directly into the central compartment amount Ac
depot(target=Ac)

EQUATION:
odeType = stiff
Ac_0 = 0

; Central concentration (defined before use)
Cc = Ac/V

; Linear + Michaelis-Menten elimination, written on amounts
ddt_Ac = -k*Ac - Vm*Cc/(Km + Cc)

OUTPUT:
output = {Cc}
