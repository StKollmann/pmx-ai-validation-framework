[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; zero-order input into the absorption compartment Ad
depot(target=Ad, Tk0)

EQUATION:
t_0 = 0
Ad_0 = 0
Ac_0 = 0
Ap_0 = 0

; define concentrations first
Cc = Ac/V
Cp = Ap/V2

ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - Cl*Cc - Q*Cc + Q*Cp
ddt_Ap = Q*Cc - Q*Cp

OUTPUT:
output = {Cc}
