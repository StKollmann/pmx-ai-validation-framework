DESCRIPTION: Target cell limited viral dynamics model with eclipse phase and loss of virions by infection

[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff
t_0 = 0

; infection rate constant derived from R0
beta = R0*c*delta/(p*T0)

T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

ddt_T = -beta*T*V
ddt_E = beta*T*V - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - beta*T*V

LV = max(log10(V), -5)

OUTPUT:
output = {LV}
