DESCRIPTION: K-PD model with turnover response model, inhibition of production (Imax sigmoid)

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
depot(target=Ak)

EQUATION:
odeType = stiff
t_0 = 0
Ak_0 = 0
R_0 = R0

kin = R0*kdeg
INH = Imax*Ak^gamma/(IC50^gamma + Ak^gamma)

ddt_Ak = -k*Ak
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
