DESCRIPTION:
Target cell limited viral dynamics model with eclipse phase.
Loss of virions due to infection of target cells is included.

[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate constant derived from the basic reproduction number
beta = R0*c*delta/(p*T0)

t_0 = 0
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

ddt_T = -beta*T*V
ddt_E = beta*T*V - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - beta*T*V

LV = max(log10(max(V,1e-10)), -5)

OUTPUT:
output = {LV}
table = {T, E, I, V}
