[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff
beta = R0*c*delta/(p*T0)
T_0 = T0
Ie_0 = 0
Ip_0 = 0
V_0 = 10^V0
ddt_T = -beta*T*V
ddt_Ie = beta*T*V - kI*Ie
ddt_Ip = kI*Ie - delta*Ip
ddt_V = p*Ip - c*V - beta*T*V
Vsat = max(V, 1e-5)
LV = log(Vsat)/log(10)

OUTPUT:
output = {LV}
