[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff
beta = R0*delta*c/(T0*(p-R0*delta))
V_0 = 10^V0
T_0 = T0
I1_0 = 0
I2_0 = 0
ddt_T = -beta*V*T
ddt_I1 = beta*V*T - kI*I1
ddt_I2 = kI*I1 - delta*I2
ddt_V = p*I2 - c*V - beta*V*T
if V > 1e-5
  LV = log(V)/log(10)
else
  LV = -5
end

OUTPUT:
output = {LV}
