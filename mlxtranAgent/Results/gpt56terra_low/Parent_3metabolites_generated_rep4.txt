[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
depot(target=Ad1, adm=1, p=0.1284)
depot(target=Ad2, adm=2, p=0.5952)
depot(target=Ad3, adm=3, p=0.2381)
depot(target=Ad4, adm=4, p=0.0383)

EQUATION:
odeType = stiff

Cp = Ac/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

k10 = Cl/V1
k12 = Q/V1
k21 = Q/V2
km20 = Clm2/V1
km30 = Clm3/V1
km40 = Clm4/V1

ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

ddt_Ac = ka1*Ad1 - (k10 + k12 + kpm2 + kpm3 + kpm4)*Ac + k21*Ap
ddt_Ap = k12*Ac - k21*Ap

ddt_Am2 = ka2*Ad2 + kpm2*Ac - km20*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ac - km30*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ac - km40*Am4

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
