[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff
Prol_0 = kdeg*Circ0/ktr
Circ_0 = Circ0
Cc = Ac/V
drugEffect = Cc/(EC50 + Cc)
feedback = (Circ0/Circ)^beta
ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*Prol*feedback*(1-drugEffect) - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ
CN = Circ

OUTPUT:
output = {Cc, CN}
