[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Cc = Ac/V
Inhib = Cc/(EC50 + Cc)
Feedback = (Circ0/CN)^beta
Prol_0 = kdeg*Circ0/ktr
CN_0 = Circ0

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*Prol*Feedback*(1 - Inhib) - ktr*Prol
ddt_CN = ktr*Prol - kdeg*CN

OUTPUT:
output = {Cc, CN}
