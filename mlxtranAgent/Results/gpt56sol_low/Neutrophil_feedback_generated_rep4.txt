[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = kdeg*Circ0/ktr
CN_0 = Circ0

Cc = Ac/V
inhib = EC50/(EC50 + Cc)
feedback = (Circ0/CN)^beta
production = kprod*Prol*inhib*feedback

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = production - ktr*Prol
ddt_CN = ktr*Prol - kdeg*CN

OUTPUT:
output = {Cc, CN}
