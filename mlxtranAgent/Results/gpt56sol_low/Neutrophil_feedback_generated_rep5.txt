[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = kdeg*Circ0/ktr
Circ_0 = Circ0

Cc = Ac/V
drug_inhibition = Cc/(EC50 + Cc)
feedback = (Circ0/Circ)^beta
proliferation = kprod*(1 - drug_inhibition)*feedback*Prol

; First-order absorption is handled by depot(target=Ac, ka)
ddt_Ac = -(Cl/V)*Ac
ddt_Prol = proliferation - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
