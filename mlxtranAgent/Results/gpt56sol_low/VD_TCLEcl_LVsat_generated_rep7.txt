[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

beta = R0*delta*c/(T0*(p - R0*delta))

T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

ddt_T = -beta*T*V
ddt_E = beta*T*V - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - beta*T*V

V_display = max(V, 1e-5)
LV = log(V_display)/log(10)

OUTPUT:
output = {LV}
