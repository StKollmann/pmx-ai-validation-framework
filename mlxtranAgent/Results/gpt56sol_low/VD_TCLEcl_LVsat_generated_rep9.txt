[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

beta = R0*c*delta/(T0*(p - R0*delta))
inf = beta*T*V

ddt_T = -inf
ddt_E = inf - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - inf

Vdisplay = max(V, 1e-5)
LV = log10(Vdisplay)

OUTPUT:
output = {LV}
