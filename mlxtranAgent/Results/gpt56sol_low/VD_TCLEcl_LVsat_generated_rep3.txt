[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

beta = R0*c*delta/(p*T0)
infection = beta*T*V

ddt_T = -infection
ddt_E = infection - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - infection

Vsafe = max(V, 1e-5)
LVraw = log(Vsafe)/log(10)
LV = max(LVraw, -5)

OUTPUT:
output = {LV}
