[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

beta = R0*c*delta/(p*T0)
infRate = beta*T*V

ddt_T = -infRate
ddt_E = infRate - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - infRate

Vsafe = max(V, 1e-5)
LVraw = log10(Vsafe)
LV = max(LVraw, -5)

OUTPUT:
output = {LV}
