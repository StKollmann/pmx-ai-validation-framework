[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; derived infection rate constant from R0 (target-cell limited, quasi-steady-state relation)
beta = R0*c*delta/(p*T0)

t_0 = 0
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

; log10 viral load with lower saturation at -5
LV = max(log10(max(V,1e-10)), -5)

OUTPUT:
output = {LV}
