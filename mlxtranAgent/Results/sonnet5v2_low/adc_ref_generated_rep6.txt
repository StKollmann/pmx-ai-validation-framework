[LONGITUDINAL]
input = {k12, k21, V, k10, V_payload, k_payload}

PK:
; ADC disposition: central + peripheral compartments, IV administration
Cc_ADC = pkmodel(k12, k21, k=k10, V)

EQUATION:
; DAR (drug-to-antibody ratio) is constant
DAR = 4

; Amount of ADC in central compartment (derived from concentration)
Ac_ADC = Cc_ADC*V

; Payload released proteolytically from ADC in central compartment
ddt_A_payload = DAR*k10*Ac_ADC - k_payload*A_payload

Cc_payload = A_payload/V_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
