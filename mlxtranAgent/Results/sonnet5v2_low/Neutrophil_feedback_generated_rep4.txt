[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ad, ka)

EQUATION:
odeType = stiff

; PK
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; Initial conditions for PD (steady-state, system starts at baseline)
Prolif_0 = Circ0
Circ_0 = Circ0

; Drug inhibition on proliferation (Emax-type inhibitory term)
Inhib = 1 - Cc/(EC50 + Cc)

; Feedback from circulating neutrophils onto proliferation compartment
Feedback = (Circ0/Circ)^beta

ddt_Prolif = kprod*Prolif*Inhib*Feedback - ktr*Prolif
ddt_Circ = ktr*Prolif - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
