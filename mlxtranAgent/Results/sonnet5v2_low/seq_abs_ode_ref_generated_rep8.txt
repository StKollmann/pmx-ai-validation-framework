[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
depot(target=Ad, Tk0=Tk0)
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
k12 = Q/V
k21 = Q/V2
peripheral(k12, k21)

EQUATION:
Cc = Ac/V
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac - k12*Ac + k21*Ap
ddt_Ap = k12*Ac - k21*Ap

OUTPUT:
output = {Cc}
