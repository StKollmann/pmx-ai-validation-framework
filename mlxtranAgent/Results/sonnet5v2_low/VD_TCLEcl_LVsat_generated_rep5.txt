[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate constant derived from R0 (R0 = beta*p*T0/(delta*c))
beta = R0*delta*c/(p*T0)

; initial conditions
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; target cell limited model with eclipse phase
; virions are lost upon infecting a target cell (loss term -beta*T*V in ddt_V)
ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

; log10 viral load with lower saturation at -5
LV = max(log10(V), -5)

OUTPUT:
output = {LV}
