[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ad, ka)

EQUATION:
odeType = stiff

; PK
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac
Cc = Ac/V

; PD - initial conditions (steady-state assumption)
Prol_0 = kdeg*Circ0/ktr
Circ_0 = Circ0

; Drug inhibition on proliferation (Emax model)
Inh = Cc/(EC50 + Cc)

; Feedback from circulating compartment on proliferation
Fb = (Circ0/Circ)^beta

ddt_Prol = kprod*Prol*(1 - Inh)*Fb - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
