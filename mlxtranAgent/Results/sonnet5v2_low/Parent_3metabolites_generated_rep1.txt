[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; Fixed first-pass fractions of the dose (sum to 1)
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

; Depot compartments receiving the dose fractions (adm=1, same dosing event)
depot(target=Ad1, p=F1, adm=1)
depot(target=Ad2, p=F2, adm=1)
depot(target=Ad3, p=F3, adm=1)
depot(target=Ad4, p=F4, adm=1)

EQUATION:
odeType=stiff

; Absorption from depots
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; Parent drug: 2-compartment disposition + conversion to metabolites
ddt_Ac   = ka1*Ad1 - (Cl/V1)*Ac - (Q/V1)*Ac + (Q/V2)*Aper - kpm2*Ac - kpm3*Ac - kpm4*Ac
ddt_Aper = (Q/V1)*Ac - (Q/V2)*Aper

; Metabolites: one compartment each, volume fixed to V1, receiving direct first-pass dose + parent conversion
ddt_Am2 = ka2*Ad2 + kpm2*Ac - (Clm2/V1)*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ac - (Clm3/V1)*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ac - (Clm4/V1)*Am4

Cp  = Ac/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
