[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
Cc = pkmodel(ka, V, Cl)

EQUATION:
; Rss correlated with R0 through a linear relationship
Rss = intercept + slope*R0
; kin derived so that in absence of drug, R reaches Rss at steady state
kin = kout*Rss

; initial condition: PD not at steady-state at t=0
R_0 = R0

; turnover model with stimulation of production by the drug
ddt_R = kin*(1 + Emax*Cc/(EC50+Cc)) - kout*R

OUTPUT:
output = {Cc, R}
