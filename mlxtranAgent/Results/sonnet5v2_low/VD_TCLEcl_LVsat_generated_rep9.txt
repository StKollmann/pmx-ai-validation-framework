[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; beta derived from R0 = beta*p*T0/(c*delta) => beta = R0*c*delta/(p*T0)
beta = R0*c*delta/(p*T0)

T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

ddt_T = -beta*T*V
ddt_E =  beta*T*V - kI*E
ddt_I =  kI*E - delta*I
ddt_V =  p*I - c*V - beta*T*V

LV = max(log10(V), -5)

OUTPUT:
output = {LV}
