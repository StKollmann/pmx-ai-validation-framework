DESCRIPTION:
Parent drug (2 compartments) with 3 metabolites (1 compartment each).
First-pass effect: the oral dose is split into 4 fixed fractions, absorbed
into the parent central compartment (F1) and directly into the metabolite
compartments (F2, F3, F4). Additional systemic conversion parent -> metabolites
via first-order rate constants kpm2, kpm3, kpm4.

[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; fixed dose fractions (first-pass effect)
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

; four depot compartments receiving the same dose, split by fractions
depot(target=Ad1, p=F1)
depot(target=Ad2, p=F2)
depot(target=Ad3, p=F3)
depot(target=Ad4, p=F4)

EQUATION:
odeType = stiff

; initial conditions
t_0  = 0
Ad1_0 = 0
Ad2_0 = 0
Ad3_0 = 0
Ad4_0 = 0
Ap_0  = 0
Ap2_0 = 0
Am2_0 = 0
Am3_0 = 0
Am4_0 = 0

; metabolite volumes fixed to parent central volume
Vm2 = V1
Vm3 = V1
Vm4 = V1

k   = Cl/V1
k12 = Q/V1
k21 = Q/V2
km2 = Clm2/Vm2
km3 = Clm3/Vm3
km4 = Clm4/Vm4

; absorption compartments
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; parent central and peripheral compartments
ddt_Ap  = ka1*Ad1 - k*Ap - k12*Ap + k21*Ap2 - (kpm2 + kpm3 + kpm4)*Ap
ddt_Ap2 = k12*Ap - k21*Ap2

; metabolites
ddt_Am2 = ka2*Ad2 + kpm2*Ap - km2*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ap - km3*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ap - km4*Am4

; concentrations
Cp  = Ap/V1
Cm2 = Am2/Vm2
Cm3 = Am3/Vm3
Cm4 = Am4/Vm4

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
