DESCRIPTION:
K-PD model: virtual PK compartment with first-order degradation (no volume,
no absorption) driving an indirect response (turnover) model with inhibition
of the production (proliferation) rate through a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; doses enter the virtual KPD compartment A
depot(target=A)

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
A_0 = 0
R_0 = R0

; production rate at baseline (steady state: kin = R0*kdeg)
kin = R0*kdeg

; virtual KPD amount with first-order degradation
ddt_A = -k*A

; sigmoidal Imax inhibition of the production rate
INH = Imax*A^gamma/(IC50^gamma + A^gamma)

; turnover with inhibition of production
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
