DESCRIPTION: One-compartment IV model written with ODEs, outputs concentration and AUC.

[LONGITUDINAL]
input = {k, V}

PK:
; Doses (IV bolus or infusion) enter the central compartment amount Ac
depot(target=Ac)

EQUATION:
t_0 = 0
Ac_0 = 0
AUC_0 = 0

ddt_Ac = -k*Ac
Cc = Ac/V
ddt_AUC = Cc

OUTPUT:
output = {Cc, AUC}
