DESCRIPTION:
K-PD model: virtual PK compartment with first-order degradation (rate k),
driving an indirect response (turnover) model with inhibition of the
production (proliferation) rate through a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual KPD compartment receiving the doses
depot(target=A)

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
A_0 = 0
R_0 = R0

; production rate at baseline
kin = R0*kdeg

; sigmoidal Imax inhibition driven by the virtual amount A
INH = Imax*A^gamma/(A^gamma + IC50^gamma)

; virtual PK compartment (degradation only)
ddt_A = -k*A

; turnover response with inhibition of production
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
