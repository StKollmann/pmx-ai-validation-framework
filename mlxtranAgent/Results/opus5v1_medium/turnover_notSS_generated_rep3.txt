[LONGITUDINAL]
DESCRIPTION: Sequential PK-PD turnover model with stimulation of the production by the drug.
One-compartment PK with first-order absorption (PK parameters as regressors).
Indirect response PD, not at steady state at t=0.

input = {ka, V, Cl, R0, Rss, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
Cc = pkmodel(ka, V, Cl)

EQUATION:
odeType = stiff
t_0 = 0
R_0 = R0

kin = Rss*kout
E = Emax*Cc/(EC50 + Cc)
ddt_R = kin*(1 + E) - kout*R

OUTPUT:
output = {Cc, R}
