DESCRIPTION: PKPD model - 1 compartment PK with first-order absorption and lag time, linear elimination (PK macros); turnover PD model with inhibition of production.

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}
Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; One-compartment model, first-order absorption with lag time, linear elimination
Cc = pkmodel(Tlag, ka, V, Cl)

EQUATION:
odeType = stiff
; Turnover model with inhibition of production
t_0 = 0
E_0 = Rin/kout
inh = 1 - Cc/(IC50 + Cc)
ddt_E = Rin*inh - kout*E

OUTPUT:
output = {Cc, E}
