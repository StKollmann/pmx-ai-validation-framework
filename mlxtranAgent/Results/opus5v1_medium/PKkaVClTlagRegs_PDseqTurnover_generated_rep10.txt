DESCRIPTION:
PKPD model: 1-compartment PK with first-order absorption and lag time, linear
elimination (macros). PD: indirect response model with inhibition of production.

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}
Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; PK model with macros
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
absorption(cmt=1, ka, Tlag)
elimination(cmt=1, Cl)

EQUATION:
odeType = stiff
; Baseline of the turnover model
E_0 = Rin/kout
; Inhibition of the production
ddt_E = Rin*(1 - Cc/(IC50 + Cc)) - kout*E

OUTPUT:
output = {Cc, E}
