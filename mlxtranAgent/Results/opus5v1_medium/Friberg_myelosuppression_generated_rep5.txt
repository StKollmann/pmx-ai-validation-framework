DESCRIPTION: Friberg semi-mechanistic myelosuppression model with 1-cmt oral PK

[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
; first-order absorption handled by depot macro (implicit absorption compartment)
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; ---- initial conditions ----
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0
Transit1_0 = Circ0
Transit2_0 = Circ0
Transit3_0 = Circ0
Circ_0 = Circ0

; ---- PK ----
ddt_Ac = -(Cl/V)*Ac
Cc = Ac/V

; ---- PD (Friberg) ----
ktr = 4/MTT
kprol = ktr
kcirc = ktr
Edrug = Emax*Cc/(EC50 + Cc)
FB = (Circ0/max(Circ,1e-6))^gam

ddt_Prol = kprol*Prol*(1 - Edrug)*FB - ktr*Prol
ddt_Transit1 = ktr*Prol - ktr*Transit1
ddt_Transit2 = ktr*Transit1 - ktr*Transit2
ddt_Transit3 = ktr*Transit2 - ktr*Transit3
ddt_Circ = ktr*Transit3 - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
