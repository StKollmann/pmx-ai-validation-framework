DESCRIPTION: Simeoni TGI model driven by cisplatin + pemetrexed exposure

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; --- fixed PK parameters ---
ka_cis = 66.5421
k_cis  = 0.2868
Vd_cis = 65.1131
ka_pem = 28.6
k_pem  = 2.1328
Vd_pem = 102.7673

; --- Cisplatin: compartment 2, administration 2 ---
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; --- Pemetrexed: compartment 3, administration 1 ---
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; --- fixed switch parameter ---
psi = 20

; --- initial conditions ---
TS_0  = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

; --- total tumor size ---
TotalTS = TS + TS1 + TS2 + TS3

; --- drug exposure ---
EXPOSURE = C_cis + C_pem

; --- Simeoni growth function with exponential-to-linear switch ---
growth = kge*TS / (1 + (kge/kgl*TotalTS)^psi)^(1/psi)

; --- log-kill effect ---
kill = kkill*EXPOSURE*TS

; --- ODEs ---
ddt_TS  = growth - kill
ddt_TS1 = kill - TS1/tau
ddt_TS2 = (TS1 - TS2)/tau
ddt_TS3 = (TS2 - TS3)/tau

OUTPUT:
output = {TotalTS}
