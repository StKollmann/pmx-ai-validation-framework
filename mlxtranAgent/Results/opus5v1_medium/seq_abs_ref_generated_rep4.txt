[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; transfer rate constants for the peripheral compartment
k23 = Q/V
k32 = Q/V2

; depot compartment with zero-order input
compartment(cmt=1, amount=Ad)
absorption(cmt=1, Tk0)

; central compartment
compartment(cmt=2, amount=Ac, volume=V, concentration=Cc)

; first-order transfer from depot to central
transfer(from=1, to=2, kt=ka)

; peripheral compartment linked to central (cmt 2)
peripheral(k23, k32)

; linear elimination from central
elimination(cmt=2, Cl)

OUTPUT:
output = {Cc}
