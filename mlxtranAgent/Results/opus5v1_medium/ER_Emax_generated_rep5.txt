DESCRIPTION:
Exposure-response model with average concentration (Cavg) read from the dataset
as a regressor, linked to the response through an effect compartment (Ce) and
an Emax model.

[LONGITUDINAL]
input = {E0, ke0, Emax, EC50, Cavg}
Cavg = {use=regressor}

EQUATION:
odeType = stiff

; Effect compartment driven by the exposure metric Cavg
t_0 = 0
Ce_0 = 0
ddt_Ce = ke0 * (Cavg - Ce)

; Emax response
E = E0 + Emax * Ce / (EC50 + Ce)

OUTPUT:
output = {E}
table = {Ce}
