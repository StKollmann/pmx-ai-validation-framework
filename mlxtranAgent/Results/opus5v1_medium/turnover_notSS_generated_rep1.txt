[LONGITUDINAL]
input = {ka, V, Cl, R0, Rss, kout, Emax, EC50}

ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; One-compartment model with first-order absorption
Cc = pkmodel(ka, V, Cl)

EQUATION:
odeType = stiff

; Initial condition (PD not at steady state at t=0)
t_0 = 0
R_0 = R0

; Zero-order production rate derived from the steady-state value Rss
kin = Rss * kout

; Stimulation of the production by the drug
E = Emax * Cc / (EC50 + Cc)

ddt_R = kin * (1 + E) - kout * R

OUTPUT:
output = {Cc, R}
