DESCRIPTION: One-compartment PK with first-order absorption and linear elimination,
coupled to a two-subpopulation neutrophil turnover model (proliferating and circulating)
with drug inhibition of proliferation and feedback from circulating neutrophils.

[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; Initial conditions (baseline steady state)
t_0 = 0
Ac_0 = 0
Prol_0 = kdeg*Circ0/ktr
CN_0 = Circ0

; PK
Cc = Ac/V

; Drug effect: inhibition of proliferation (Emax with Imax = 1)
INH = 1 - Cc/(EC50 + Cc)

; Feedback from circulating neutrophils
FB = (Circ0/max(CN, 1e-6))^beta

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*Prol*INH*FB - ktr*Prol
ddt_CN = ktr*Prol - kdeg*CN

OUTPUT:
output = {Cc, CN}
table = {Prol}
