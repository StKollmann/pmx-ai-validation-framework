DESCRIPTION:
K-PD model: virtual PK compartment with first-order degradation (no volume,
no absorption) driving a turnover (indirect response) PD model with
inhibition of the production (proliferation) rate by a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual KPD compartment receiving the doses
depot(target=A)

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
A_0 = 0
R_0 = R0

; virtual "concentration" = amount in the KPD compartment
ddt_A = -k*A

; sigmoidal Imax inhibition of the production rate
INH = Imax*A^gamma/(IC50^gamma + A^gamma)

; turnover model : production inhibited, first-order loss
kin = R0*kdeg
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
table = {A}
