DESCRIPTION:
One-compartment PK with first-order absorption and linear elimination,
coupled to a turnover PD model with proliferating and circulating neutrophils.

[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
; doses enter a hidden depot, absorbed into Ac at rate ka
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; --- initial conditions ---
t_0 = 0
Ac_0 = 0
P_0 = kdeg*Circ0/ktr
CN_0 = Circ0

; --- PK ---
Cc = Ac/V
ddt_Ac = -(Cl/V)*Ac

; --- drug effect and feedback (defined before use) ---
INH = 1 - Cc/(EC50 + Cc)
FDBK = (Circ0/max(CN,1e-6))^beta

; --- PD turnover ---
ddt_P  = kprod*P*INH*FDBK - ktr*P
ddt_CN = ktr*P - kdeg*CN

OUTPUT:
output = {Cc, CN}
