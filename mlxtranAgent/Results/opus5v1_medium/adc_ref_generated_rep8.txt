DESCRIPTION:
ADC model: 2-compartment ADC disposition (IV) with proteolytic degradation
releasing cytotoxic payload (DAR = 4) into a one-compartment payload model.

[LONGITUDINAL]
input = {k12, k21, V, k10, V_payload, k_payload}

PK:
; --- ADC disposition written with macros ---
compartment(cmt=1, amount=Ac_ADC, volume=V, concentration=Cc_ADC)
iv(cmt=1, adm=1)
elimination(cmt=1, k=k10)
peripheral(k12, k21)

EQUATION:
odeType = stiff
DAR = 4
t_0 = 0
A_payload_0 = 0

; payload released by proteolytic degradation of ADC in central compartment
release = DAR * k10 * Ac_ADC

ddt_A_payload = release - k_payload * A_payload

Cc_payload = A_payload / V_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
