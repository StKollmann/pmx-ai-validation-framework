[LONGITUDINAL]
input = {k, V}

PK:
; IV administration (bolus or infusion) directly into the central compartment
depot(target = Ac)

EQUATION:
t_0 = 0
Ac_0 = 0
AUC_0 = 0

; One-compartment disposition with first-order elimination
ddt_Ac = -k*Ac
Cc = Ac/V
; Cumulative area under the concentration-time curve
ddt_AUC = Cc

OUTPUT:
output = {Cc, AUC}
