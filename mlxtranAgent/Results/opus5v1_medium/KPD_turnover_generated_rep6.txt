DESCRIPTION:
K-PD model: virtual kinetics compartment with first-order degradation (no volume,
no absorption) driving a turnover (indirect response) PD model with inhibition of
the production (proliferation) rate through a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual KPD compartment receiving the doses
depot(target=A)

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
A_0 = 0
R_0 = R0

; virtual concentration (amount, no volume) with first-order degradation
ddt_A = -k*A

; sigmoidal Imax inhibition of the production rate
INH = Imax*max(A,0)^gamma / (IC50^gamma + max(A,0)^gamma)

; turnover model : kin = R0*kdeg, kout = kdeg
ddt_R = R0*kdeg*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
table = {A, INH}
