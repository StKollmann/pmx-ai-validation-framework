DESCRIPTION:
One-compartment PK model with intravenous administration and time-dependent
(linearly changing) clearance: CL(t) = CL0 + slope*t

[LONGITUDINAL]
input = {V, CL0, slope}

PK:
; dose enters the central compartment directly (IV bolus/infusion)
depot(target=Ac)

EQUATION:
odeType = stiff
t_0 = 0
Ac_0 = 0

; time-varying clearance (bounded below by 0 to avoid negative elimination)
CLt = max(CL0 + slope*t, 0)
k = CLt/V

ddt_Ac = -k*Ac

Cc = Ac/V

OUTPUT:
output = {Cc}
table = {CLt}
