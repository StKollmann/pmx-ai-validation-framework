DESCRIPTION:
Friberg semi-mechanistic myelosuppression model with 1-compartment PK
(first-order absorption via depot macro, linear elimination).

[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
; doses enter a hidden absorption compartment, absorbed into Ac at rate ka
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; ---- initial conditions ----
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0
Tr1_0 = Circ0
Tr2_0 = Circ0
Tr3_0 = Circ0
Circ_0 = Circ0

; ---- PK ----
ke = Cl/V
Cc = Ac/V
ddt_Ac = -ke*Ac

; ---- PD (Friberg) ----
ktr = 4/MTT
kprol = ktr
kcirc = ktr

; drug effect: inhibition of proliferation (Emax model)
Edrug = Emax*Cc/(EC50 + Cc)

; homeostatic feedback
FB = (Circ0/max(Circ,1e-10))^gam

ddt_Prol = kprol*Prol*(1 - Edrug)*FB - ktr*Prol
ddt_Tr1  = ktr*Prol - ktr*Tr1
ddt_Tr2  = ktr*Tr1  - ktr*Tr2
ddt_Tr3  = ktr*Tr2  - ktr*Tr3
ddt_Circ = ktr*Tr3  - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
