DESCRIPTION:
Target cell limited viral dynamics model with an eclipse phase.
Loss of free virions due to infection of target cells is included.
State variables:
  T  : target cells
  I1 : infected cells in eclipse phase (not yet producing virus)
  I2 : productively infected cells
  V  : free virus

[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate constant derived from the basic reproduction number
beta = R0*c*delta/(p*T0)

; initial conditions
t_0  = 0
T_0  = T0
I1_0 = 0
I2_0 = 0
V_0  = 10^V0

; system of ODEs
ddt_T  = -beta*T*V
ddt_I1 =  beta*T*V - kI*I1
ddt_I2 =  kI*I1 - delta*I2
ddt_V  =  p*I2 - c*V - beta*T*V

; log10 viral load with lower saturation
LV = max(log10(max(V, 1e-10)), -5)

OUTPUT:
output = {LV}
table = {T, I1, I2, V}
