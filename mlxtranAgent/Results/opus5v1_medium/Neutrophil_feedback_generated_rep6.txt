DESCRIPTION: One-compartment PK with first-order absorption and linear elimination,
coupled to a two-subpopulation neutrophil turnover PD model (proliferating and circulating).

[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
; doses enter a hidden depot, absorbed into Ac at rate ka
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; --- Initial conditions ---
t_0 = 0
Ac_0 = 0
P_0 = Circ0*kdeg/ktr
CN_0 = Circ0

; --- PK ---
Cc = Ac/V
ddt_Ac = -(Cl/V)*Ac

; --- Drug effect (inhibition of proliferation) and feedback ---
INH = 1 - Cc/(EC50 + Cc)
FB = (Circ0/max(CN,1e-6))^beta

; --- PD: turnover of neutrophils ---
ddt_P = kprod*P*INH*FB - ktr*P
ddt_CN = ktr*P - kdeg*CN

OUTPUT:
output = {Cc, CN}
table = {P}
