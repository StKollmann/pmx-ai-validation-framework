DESCRIPTION:
Simeoni TGI model with cell distribution death delay (3 transit compartments),
driven by the sum of cisplatin and pemetrexed concentrations.

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau,
         ka_cis, k_cis, Vd_cis,
         ka_pem, k_pem, Vd_pem}

PK:
; ---- Cisplatin PK (compartment 2, administration 2) ----
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; ---- Pemetrexed PK (compartment 3, administration 1) ----
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; initial conditions
TSp_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

psi = 20

; total tumor size (must be defined before use)
TotalTS = TSp + TS1 + TS2 + TS3

; drug exposure
EXPOSURE = C_cis + C_pem

; Simeoni growth function (exponential then linear switch)
growth = kge*TSp / (1 + (kge/kgl*TotalTS)^psi)^(1/psi)

; log-kill treatment effect
kill = kkill*EXPOSURE*TSp

ddt_TSp = growth - kill
ddt_TS1 = kill - TS1/tau
ddt_TS2 = (TS1 - TS2)/tau
ddt_TS3 = (TS2 - TS3)/tau

OUTPUT:
output = {TotalTS}
