DESCRIPTION:
Simeoni tumor growth inhibition model driven by the sum of cisplatin and
pemetrexed concentrations, with a cell distribution (transit) death model.

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; --- Cisplatin PK (fixed parameters), cmt=2, adm=2 ---
ka_cis = 66.5421
k_cis  = 0.2868
Vd_cis = 65.1131
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; --- Pemetrexed PK (fixed parameters), cmt=3, adm=1 ---
ka_pem = 28.6
k_pem  = 2.1328
Vd_pem = 102.7673
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; initial conditions
TS_0  = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

psi = 20

; drug exposure
EXPOSURE = C_cis + C_pem

; killing rate (log-kill hypothesis, linear in exposure)
kill = kkill * EXPOSURE * TS

; Simeoni growth: exponential then linear
growth = kge * TS / (1 + (kge/kgl * TS)^psi)^(1/psi)

ddt_TS  = growth - kill
ddt_TS1 = kill - TS1/tau
ddt_TS2 = (TS1 - TS2)/tau
ddt_TS3 = (TS2 - TS3)/tau

TotalTS = TS + TS1 + TS2 + TS3

OUTPUT:
output = {TotalTS}
table = {C_cis, C_pem, EXPOSURE}
