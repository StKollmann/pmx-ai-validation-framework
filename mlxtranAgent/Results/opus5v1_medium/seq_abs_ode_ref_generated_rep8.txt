[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; Zero-order input of the dose into the absorption compartment Ad
depot(target=Ad, Tk0)

EQUATION:
odeType = stiff

; Initial conditions
Ad_0 = 0
Ac_0 = 0
Ap_0 = 0

; First-order absorption from Ad to central compartment
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac - (Q/V)*Ac + (Q/V2)*Ap
ddt_Ap = (Q/V)*Ac - (Q/V2)*Ap

Cc = Ac/V

OUTPUT:
output = {Cc}
