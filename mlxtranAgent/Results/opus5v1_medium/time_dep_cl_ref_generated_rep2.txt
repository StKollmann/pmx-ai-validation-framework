DESCRIPTION:
One-compartment model with IV administration and time-dependent (linear) clearance.
CL(t) = CL0 + slope*t, constrained to be non-negative.

[LONGITUDINAL]
input = {V, CL0, slope}

PK:
; dose goes directly into the central compartment amount Ac
depot(target=Ac)

EQUATION:
odeType = stiff
t_0 = 0
Ac_0 = 0

; time-varying clearance (linear change over time)
CLt = max(CL0 + slope*t, 0)
k = CLt/V

ddt_Ac = -k*Ac

Cc = Ac/V

OUTPUT:
output = {Cc}
table = {CLt}
