[LONGITUDINAL]
input = {ka, V, Cl, R0, Rss, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; One-compartment model with first-order absorption (PK parameters as regressors)
Cc = pkmodel(ka, V, Cl)

EQUATION:
odeType = stiff
t_0 = 0
R_0 = R0

; production rate constant, defined so that the drug-free steady state is Rss
kin = kout*Rss

; stimulation of the production by the drug
Estim = Emax*Cc/(EC50 + Cc)

ddt_R = kin*(1 + Estim) - kout*R

OUTPUT:
output = {Cc, R}
