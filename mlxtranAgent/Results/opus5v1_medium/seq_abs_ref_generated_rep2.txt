DESCRIPTION: Two-compartment PK model with sequential zero-order then first-order absorption (macros only)

[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; inter-compartmental rate constants
k23 = Q/V
k32 = Q/V2

; depot compartment filled by zero-order input
compartment(cmt=1, amount=Ad)
absorption(cmt=1, Tk0)

; first-order transfer from depot to central compartment
transfer(from=1, to=2, kt=ka)

; central compartment
compartment(cmt=2, amount=Ac, volume=V, concentration=Cc)
elimination(cmt=2, Cl)

; peripheral compartment
peripheral(k23, k32)

OUTPUT:
output = {Cc}
