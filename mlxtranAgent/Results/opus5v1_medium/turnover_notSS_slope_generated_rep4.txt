DESCRIPTION:
; Sequential PK-PD model
; PK: one-compartment, first-order absorption, PK parameters given as regressors
; PD: turnover (indirect response) model with stimulation of the production
; Not at steady state at t=0: R(0) = R0, and Rss = intercept + slope*R0

[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}

ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; one-compartment model with first-order absorption
Cc = pkmodel(ka, V, Cl)

EQUATION:
odeType = stiff

; steady-state response and corresponding production rate
Rss = intercept + slope*R0
kin = kout*Rss

; initial condition (system not at steady state at t=0)
t_0 = 0
R_0 = R0

; stimulation of the production by the drug
Stim = Emax*Cc/(EC50 + Cc)
ddt_R = kin*(1 + Stim) - kout*R

OUTPUT:
output = {Cc, R}
