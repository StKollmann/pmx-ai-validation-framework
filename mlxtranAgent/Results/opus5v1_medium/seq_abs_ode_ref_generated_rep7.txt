[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; zero-order input of the dose into the absorption compartment Ad
depot(target=Ad, Tk0)

EQUATION:
odeType = stiff
t_0 = 0
Ad_0 = 0
Ac_0 = 0
Ap_0 = 0

; first-order absorption from Ad to central compartment, 2-cmt disposition
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac - (Q/V)*Ac + (Q/V2)*Ap
ddt_Ap = (Q/V)*Ac - (Q/V2)*Ap

Cc = Ac/V

OUTPUT:
output = {Cc}
