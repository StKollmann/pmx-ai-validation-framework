DESCRIPTION:
ADC model: 2-compartment ADC disposition (IV), proteolytic release of payload
from the ADC central compartment (DAR = 4), payload with 1-compartment disposition.

[LONGITUDINAL]
input = {k12, k21, V, k10, V_payload, k_payload}

PK:
; --- ADC disposition written with macros ---
compartment(cmt=1, amount=Ac_ADC, volume=V, concentration=Cc_ADC)
iv(cmt=1, adm=1)
peripheral(k12, k21)
elimination(cmt=1, k=k10)

EQUATION:
odeType = stiff
DAR = 4
t_0 = 0
A_payload_0 = 0

; payload released from ADC in central compartment (proteolytic pathway only)
ddt_A_payload = DAR*k10*Ac_ADC - k_payload*A_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
