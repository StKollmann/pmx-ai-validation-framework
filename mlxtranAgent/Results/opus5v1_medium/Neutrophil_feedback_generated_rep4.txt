DESCRIPTION:
One-compartment PK with first-order absorption and linear elimination,
coupled to a turnover PD model with proliferating and circulating neutrophils.

[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
; doses enter a hidden depot compartment, absorbed into Ac at rate ka
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; --- initial conditions (system at steady state before treatment) ---
t_0 = 0
Ac_0 = 0
P_0 = Circ0*kdeg/ktr
Circ_0 = Circ0

; --- observable variables (defined before use) ---
Cc = Ac/V
CN = Circ

; --- Emax inhibition of proliferation by the drug ---
INH = 1 - Cc/(EC50 + Cc)

; --- feedback of circulating neutrophils on proliferation ---
FB = (Circ0/max(CN, 1e-6))^beta

; --- ODEs ---
ddt_Ac = -(Cl/V)*Ac
ddt_P = kprod*P*INH*FB - ktr*P
ddt_Circ = ktr*P - kdeg*Circ

OUTPUT:
output = {Cc, CN}
