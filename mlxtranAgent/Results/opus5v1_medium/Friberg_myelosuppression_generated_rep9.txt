[LONGITUDINAL]
DESCRIPTION:
Friberg semi-mechanistic model of chemotherapy-induced myelosuppression.
One-compartment PK with first-order absorption (depot macro with ka) and linear elimination.
PD with proliferating compartment, 3 transit compartments and circulating compartment.

input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
; first-order absorption handled by the depot macro
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; ---- initial conditions ----
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0
Tr1_0 = Circ0
Tr2_0 = Circ0
Tr3_0 = Circ0
Circ_0 = Circ0

; ---- PK ----
k = Cl/V
Cc = Ac/V

; ---- transit rate (3 transit compartments) ----
ktr = 4/MTT
kprol = ktr
kcirc = ktr

; ---- drug effect and feedback ----
Edrug = Emax*Cc/(EC50 + Cc)
FB = (Circ0/max(Circ, 1e-6))^gam

; ---- ODEs ----
ddt_Ac = -k*Ac
ddt_Prol = kprol*Prol*(1 - Edrug)*FB - ktr*Prol
ddt_Tr1 = ktr*Prol - ktr*Tr1
ddt_Tr2 = ktr*Tr1 - ktr*Tr2
ddt_Tr3 = ktr*Tr2 - ktr*Tr3
ddt_Circ = ktr*Tr3 - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
