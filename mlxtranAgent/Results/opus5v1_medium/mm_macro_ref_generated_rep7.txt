[LONGITUDINAL]
input = {V, k, Vm, Km}

PK:
; One-compartment model with IV bolus administration
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
iv(cmt=1, adm=1)
elimination(cmt=1, k)          ; linear elimination (first-order rate)
elimination(cmt=1, Vm, Km)     ; Michaelis-Menten elimination

OUTPUT:
output = {Cc}
