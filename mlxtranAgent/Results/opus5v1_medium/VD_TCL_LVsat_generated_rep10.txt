[LONGITUDINAL]
DESCRIPTION: Target cell limited model of viral dynamics

input = {beta, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
T_0 = T0
I_0 = 0
V_0 = V0

; system of ODEs
ddt_T = -beta*T*V
ddt_I = beta*T*V - delta*I
ddt_V = p*I - c*V

; log10 viral load with lower saturation
Vpos = max(V, 1e-10)
LV = max(log10(Vpos), -5)

OUTPUT:
output = {LV}
table = {T, I, V}
