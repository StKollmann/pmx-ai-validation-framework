[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = ktr*Circ0/(kprod-ktr)
Circ_0 = Circ0

Cc = Ac/V
CN = Circ

Inhib = Cc/(EC50 + Cc)
Feedback = (Circ0/CN)^beta
Growth = kprod*(1-Inhib)*Feedback*Prol

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = Growth - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

OUTPUT:
output = {Cc, CN}
