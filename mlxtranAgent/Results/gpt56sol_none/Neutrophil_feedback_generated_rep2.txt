[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = Circ0*kdeg/ktr
Circ_0 = Circ0

Cc = Ac/V
CN = Circ

DrugInhib = Cc/(EC50 + Cc)
Feedback = (Circ0/Circ)^beta

Production = kprod*Prol*Feedback*(1 - DrugInhib)
Transition = ktr*Prol
Loss = kdeg*Circ

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = Production - Transition
ddt_Circ = Transition - Loss

OUTPUT:
output = {Cc, CN}
