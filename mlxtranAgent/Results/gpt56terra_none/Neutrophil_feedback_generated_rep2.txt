[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff
Cc = Ac/V
Inh = Cc/(EC50 + Cc)
Feedback = (Circ0/CN)^beta
CN_0 = Circ0
Prolif_0 = Circ0*kdeg/ktr
Prod = kprod*Prolif*Feedback*(1 - Inh)
ddt_Ac = -(Cl/V)*Ac
ddt_Prolif = Prod - ktr*Prolif
ddt_CN = ktr*Prolif - kdeg*CN

OUTPUT:
output = {Cc, CN}
