[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
depot(target=Ad)

EQUATION:
odeType = stiff
; Fixed first-pass fractions: F1=0.1284, F2=0.5952, F3=0.2381, F4=0.0383
ddt_Ad = -(ka1 + ka2 + ka3 + ka4)*Ad
ddt_Ap = ka1*Ad - (Cl/V1)*Ap - (Q/V1)*Ap + (Q/V2)*App - kpm2*Ap - kpm3*Ap - kpm4*Ap
ddt_App = (Q/V1)*Ap - (Q/V2)*App
ddt_Am2 = ka2*Ad + kpm2*Ap - (Clm2/V1)*Am2
ddt_Am3 = ka3*Ad + kpm3*Ap - (Clm3/V1)*Am3
ddt_Am4 = ka4*Ad + kpm4*Ap - (Clm4/V1)*Am4
Cp = Ap/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
