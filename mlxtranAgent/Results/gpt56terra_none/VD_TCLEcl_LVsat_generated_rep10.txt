[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff
V_0 = 10^V0
T_0 = T0
E_0 = 0
I_0 = 0
beta = R0*delta*c/(p*T0)
ddt_T = -beta*V*T
ddt_E = beta*V*T - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - beta*V*T
Vsafe = max(V, 10^(-5))
LV = log10(Vsafe)

OUTPUT:
output = {LV}
