[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff
beta = R0*c*delta/(p*T0)
Vinit = 10^V0
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = Vinit
ddt_T = -beta*T*V
ddt_E = beta*T*V - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - beta*T*V
LVraw = log(V)/log(10)
if LVraw < -5
    LV = -5
else
    LV = LVraw
end

OUTPUT:
output = {LV}
