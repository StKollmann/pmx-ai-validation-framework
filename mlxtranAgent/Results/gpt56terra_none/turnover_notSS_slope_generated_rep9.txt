[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
Cc = pkmodel(ka, V, Cl)

EQUATION:
Rss = intercept + slope*R0
kin = Rss*kout
Edrug = Emax*Cc/(EC50 + Cc)
ddt_R = kin*(1 + Edrug) - kout*R
R_0 = R0

OUTPUT:
output = {Cc, R}
