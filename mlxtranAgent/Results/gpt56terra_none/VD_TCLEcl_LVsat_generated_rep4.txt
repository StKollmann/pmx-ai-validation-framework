[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff
V_0 = 10^V0
T_0 = T0
I1_0 = 0
I2_0 = 0
beta = R0*c*delta/(p*T0)
ddt_T = -beta*T*V
ddt_I1 = beta*T*V - kI*I1
ddt_I2 = kI*I1 - delta*I2
ddt_V = p*I2 - c*V - beta*T*V
if V > 1e-5
    LV = log10(V)
else
    LV = -5
end

OUTPUT:
output = {LV}
