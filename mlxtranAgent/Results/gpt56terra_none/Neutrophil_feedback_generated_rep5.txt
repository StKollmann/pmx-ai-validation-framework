[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
; Oral depot with implicit first-order absorption into the central compartment
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Cc = Ac/V
Inhib = Cc/(EC50 + Cc)
Feedback = (Circ0/CN)^beta

P_0 = Circ0
CN_0 = Circ0

ddt_P = kprod*P*(1 - Inhib)*Feedback - ktr*P
ddt_CN = ktr*P - kdeg*CN

ddt_Ac = -(Cl/V)*Ac

OUTPUT:
output = {Cc, CN}
