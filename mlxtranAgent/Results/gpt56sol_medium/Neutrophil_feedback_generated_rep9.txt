[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = kdeg*Circ0/ktr
CN_0 = Circ0

Cc = Ac/V
drug_inhibition = Cc/(EC50 + Cc)
feedback = (Circ0/CN)^beta

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*(1 - drug_inhibition)*feedback*Prol - ktr*Prol
ddt_CN = ktr*Prol - kdeg*CN

OUTPUT:
output = {Cc, CN}
