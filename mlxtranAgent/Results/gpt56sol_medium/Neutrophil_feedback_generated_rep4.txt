[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Prol_0 = kdeg*Circ0/ktr
Circ_0 = Circ0

Cc = Ac/V
CN = Circ
inhib = EC50/(EC50 + Cc)
feedback = (Circ0/CN)^beta

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*Prol*inhib*feedback - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

OUTPUT:
output = {Cc, CN}
