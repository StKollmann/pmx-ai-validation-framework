[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

depot(target=Ad1, p=F1)
depot(target=Ad2, p=F2)
depot(target=Ad3, p=F3)
depot(target=Ad4, p=F4)

EQUATION:
odeType = stiff

k10 = Cl/V1
k12 = Q/V1
k21 = Q/V2
km2 = Clm2/V1
km3 = Clm3/V1
km4 = Clm4/V1

ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

ddt_Ap = ka1*Ad1 - (k10 + k12 + kpm2 + kpm3 + kpm4)*Ap + k21*At
ddt_At = k12*Ap - k21*At

ddt_Am2 = ka2*Ad2 + kpm2*Ap - km2*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ap - km3*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ap - km4*Am4

Cp = Ap/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
