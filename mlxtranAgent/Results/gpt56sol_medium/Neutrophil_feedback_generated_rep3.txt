[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

Cc = Ac/V
feedback = (Circ0/CN)^beta
drug_effect = EC50/(EC50 + Cc)

Prol_0 = kdeg*Circ0/ktr
CN_0 = Circ0

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*Prol*feedback*drug_effect - ktr*Prol
ddt_CN = ktr*Prol - kdeg*CN

OUTPUT:
output = {Cc, CN}
