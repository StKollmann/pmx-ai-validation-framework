DESCRIPTION: Simeoni TGI model driven by cisplatin + pemetrexed exposure

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; --- Cisplatin PK (fixed parameters) ---
ka_cis = 66.5421
k_cis = 0.2868
Vd_cis = 65.1131
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; --- Pemetrexed PK (fixed parameters) ---
ka_pem = 28.6
k_pem = 2.1328
Vd_pem = 102.7673
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff
psi = 20

; initial conditions
TSp_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

; total tumor size
TotalTS = TSp + TS1 + TS2 + TS3

; exposure
EXPOSURE = C_cis + C_pem

; Simeoni growth and log-kill
growth = kge*TSp / (1 + (kge/kgl*TotalTS)^psi)^(1/psi)
kill = kkill*EXPOSURE*TSp
ktr = 1/tau

ddt_TSp = growth - kill
ddt_TS1 = kill - ktr*TS1
ddt_TS2 = ktr*TS1 - ktr*TS2
ddt_TS3 = ktr*TS2 - ktr*TS3

OUTPUT:
output = {TotalTS}
