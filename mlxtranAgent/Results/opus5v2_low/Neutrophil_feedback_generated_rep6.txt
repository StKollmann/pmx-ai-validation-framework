[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

t_0 = 0
Ac_0 = 0
PN_0 = Circ0
CN_0 = Circ0

Cc = Ac/V
inhib = 1 - Cc/(EC50 + Cc)
feedback = (Circ0/CN)^beta

ddt_Ac = -(Cl/V)*Ac
ddt_PN = kprod*PN*inhib*feedback - ktr*PN
ddt_CN = ktr*PN - kdeg*CN

OUTPUT:
output = {Cc, CN}
