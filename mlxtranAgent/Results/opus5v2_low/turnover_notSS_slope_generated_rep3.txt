[LONGITUDINAL]
input = {ka, V, Cl, R0, intercept, slope, kout, Emax, EC50}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
Cc = pkmodel(ka, V, Cl)

EQUATION:
t_0 = 0
R_0 = R0

Rss = intercept + slope*R0
kin = kout*Rss
Stim = Emax*Cc/(EC50 + Cc)
ddt_R = kin*(1 + Stim) - kout*R

OUTPUT:
output = {Cc, R}
