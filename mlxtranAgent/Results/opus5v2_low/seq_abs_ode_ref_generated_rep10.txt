[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; Zero-order input of the dose into the absorption compartment Ad over duration Tk0
depot(target=Ad, Tk0)

EQUATION:
; First-order transfer from absorption compartment to central compartment
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac - (Q/V)*Ac + (Q/V2)*Ap
ddt_Ap = (Q/V)*Ac - (Q/V2)*Ap
Cc = Ac/V

OUTPUT:
output = {Cc}
