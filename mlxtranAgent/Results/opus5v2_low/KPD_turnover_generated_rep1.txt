DESCRIPTION: K-PD model with turnover PD, inhibition of production (sigmoidal Imax)

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
depot(target=A)

EQUATION:
t_0 = 0
A_0 = 0
R_0 = R0

ddt_A = -k*A
INH = Imax*A^gamma/(A^gamma + IC50^gamma)
kin = R0*kdeg
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
