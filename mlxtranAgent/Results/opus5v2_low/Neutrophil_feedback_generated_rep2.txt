[LONGITUDINAL]
input = {ka, V, Cl, kprod, Circ0, ktr, kdeg, beta, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = stiff

; initial conditions (baseline steady state)
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0*kdeg/ktr
Circ_0 = Circ0

Cc = Ac/V
INH = 1 - Cc/(EC50 + Cc)
FB = (Circ0/max(Circ, 1e-6))^beta

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprod*Prol*INH*FB - ktr*Prol
ddt_Circ = ktr*Prol - kdeg*Circ

CN = Circ

OUTPUT:
output = {Cc, CN}
