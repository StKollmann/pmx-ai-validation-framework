[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; zero-order input (duration Tk0) into the absorption compartment Ad
depot(target=Ad, Tk0)

EQUATION:
; first-order absorption from Ad into central compartment
ddt_Ad = -ka*Ad
ddt_Ac = ka*Ad - (Cl/V)*Ac - (Q/V)*Ac + (Q/V2)*Ap
ddt_Ap = (Q/V)*Ac - (Q/V2)*Ap
Cc = Ac/V

OUTPUT:
output = {Cc}
