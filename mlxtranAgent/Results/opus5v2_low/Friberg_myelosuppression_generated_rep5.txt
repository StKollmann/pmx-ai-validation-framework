[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0
Tr1_0 = Circ0
Tr2_0 = Circ0
Tr3_0 = Circ0
Circ_state_0 = Circ0

ktr = 4/MTT
kprol = ktr
kcirc = ktr

Cc = Ac/V
Circ = max(Circ_state, 1e-6)

Edrug = Emax*Cc/(EC50+Cc)
FB = (Circ0/Circ)^gam

ddt_Ac = -(Cl/V)*Ac
ddt_Prol = kprol*Prol*(1-Edrug)*FB - ktr*Prol
ddt_Tr1 = ktr*Prol - ktr*Tr1
ddt_Tr2 = ktr*Tr1 - ktr*Tr2
ddt_Tr3 = ktr*Tr2 - ktr*Tr3
ddt_Circ_state = ktr*Tr3 - kcirc*Circ_state

OUTPUT:
output = {Cc, Circ}
