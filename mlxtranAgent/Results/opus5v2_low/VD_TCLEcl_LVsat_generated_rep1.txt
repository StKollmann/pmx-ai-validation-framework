DESCRIPTION:
Target cell limited viral dynamics model with eclipse phase and loss of virions
due to infection. Infectivity rate beta is derived from the basic reproduction
number R0.

[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
T_0 = T0
E_0 = 0
I_0 = 0
V_0 = 10^V0

; infectivity rate derived from R0
beta = R0*c*delta/(p*T0)

; ODE system
ddt_T = -beta*T*V
ddt_E = beta*T*V - kI*E
ddt_I = kI*E - delta*I
ddt_V = p*I - c*V - beta*T*V

; log10 viral load with lower saturation at -5
LV = max(log10(max(V, 1e-30)), -5)

OUTPUT:
output = {LV}
table = {T, E, I, V}
