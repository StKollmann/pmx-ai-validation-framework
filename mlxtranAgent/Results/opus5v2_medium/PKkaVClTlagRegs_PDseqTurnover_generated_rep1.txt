DESCRIPTION:
PKPD model: 1-compartment PK with first-order absorption and lag time, linear
elimination (PK macros). PD: indirect response model with inhibition of production.

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}

Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; 1-compartment model with first-order absorption and lag time
Cc = pkmodel(Tlag, ka, V, Cl)

EQUATION:
; Turnover model with inhibition of production
t_0 = 0
E_0 = Rin/kout

inh = Cc/(Cc + IC50)
ddt_E = Rin*(1 - inh) - kout*E

OUTPUT:
output = {Cc, E}
