[LONGITUDINAL]
input = {V, k, Vm, Km}

PK:
; IV bolus doses go directly into the central compartment amount Ac
depot(target=Ac)

EQUATION:
t_0 = 0
Ac_0 = 0

; define concentration BEFORE use
Cc = Ac/V

; one-compartment, parallel linear + Michaelis-Menten elimination
ddt_Ac = -k*Ac - Vm*Cc/(Km + Cc)

OUTPUT:
output = {Cc}
