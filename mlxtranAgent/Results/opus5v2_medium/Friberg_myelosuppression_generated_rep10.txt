DESCRIPTION: Friberg myelosuppression model with 1-cmt PK (first-order absorption, linear elimination)

[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
depot(target=Ac, ka)

EQUATION:
odeType = nonStiff

; --- Initial conditions ---
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0
Transit1_0 = Circ0
Transit2_0 = Circ0
Transit3_0 = Circ0
Circ_0 = Circ0

; --- Derived parameters ---
ktr = 4/MTT
kprol = ktr
kcirc = ktr

; --- PK ---
Cc = Ac/V
ddt_Ac = -(Cl/V)*Ac

; --- Drug effect (Emax inhibition of proliferation) ---
Edrug = Emax*Cc/(EC50 + Cc)
Edrug2 = min(Edrug, 0.999)

; --- Feedback ---
FB = (Circ0/max(Circ, 1e-6))^gam

; --- PD (Friberg) ---
ddt_Prol = kprol*Prol*(1 - Edrug2)*FB - ktr*Prol
ddt_Transit1 = ktr*Prol - ktr*Transit1
ddt_Transit2 = ktr*Transit1 - ktr*Transit2
ddt_Transit3 = ktr*Transit2 - ktr*Transit3
ddt_Circ = ktr*Transit3 - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
