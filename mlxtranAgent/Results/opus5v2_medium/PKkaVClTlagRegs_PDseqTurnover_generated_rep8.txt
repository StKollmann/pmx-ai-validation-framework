DESCRIPTION:
PKPD model: 1-compartment PK with first-order absorption, lag time and linear
elimination (PK macros). PD: indirect response model with inhibition of production.

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}

Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; 1-compartment model with first-order absorption and lag time
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
absorption(cmt=1, adm=1, ka, Tlag)
elimination(cmt=1, Cl)

EQUATION:
; Initial condition of the turnover model (baseline)
t_0 = 0
E_0 = Rin/kout

; Inhibition of the production rate
Inh = Cc/(Cc + IC50)
ddt_E = Rin*(1 - Inh) - kout*E

OUTPUT:
output = {Cc, E}
