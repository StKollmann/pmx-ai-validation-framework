DESCRIPTION: K-PD model (virtual dose compartment with degradation rate k) linked to an
indirect response (turnover) model with inhibition of the production rate
described by a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual K-PD compartment: doses enter A, first-order disappearance with rate k
depot(target=A)

EQUATION:
odeType = nonStiff

; initial conditions
t_0 = 0
A_0 = 0
R_0 = R0

; production rate at baseline
kin = R0*kdeg

; K-PD "concentration" (virtual amount)
ddt_A = -k*A

; sigmoidal Imax inhibition of the production rate
INH = Imax*max(A,0)^gamma/(IC50^gamma + max(A,0)^gamma)

ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
