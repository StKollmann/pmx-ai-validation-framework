DESCRIPTION:
Simeoni tumor growth inhibition model with cell distribution (3 death compartments)
driven by the sum of cisplatin and pemetrexed concentrations.

[LONGITUDINAL]
input = {TS0, kge, kgl, psi, kkill, tau, ka_cis, k_cis, Vd_cis, ka_pem, k_pem, Vd_pem}

PK:
; Cisplatin PK (cmt=2, adm=2)
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; Pemetrexed PK (cmt=3, adm=1)
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; Initial conditions
TS_0  = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

; Total tumor size
TotalTS = TS + TS1 + TS2 + TS3

; Drug exposure
EXPOSURE = C_cis + C_pem

; Simeoni growth function (exponential to linear switch)
growth = kge*TS / (1 + (kge/kgl*TotalTS)^psi)^(1/psi)

; Log-kill treatment effect
kill = kkill*EXPOSURE*TS

; Transit rate through death compartments
ktr = 1/tau

ddt_TS  = growth - kill
ddt_TS1 = kill - ktr*TS1
ddt_TS2 = ktr*TS1 - ktr*TS2
ddt_TS3 = ktr*TS2 - ktr*TS3

OUTPUT:
output = {TotalTS}
table = {C_cis, C_pem, EXPOSURE}
