DESCRIPTION:
PKPD model: 1-compartment PK with first-order absorption and lag time, linear
elimination (PK macros). PD: turnover (indirect response) model with inhibition
of production.

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}

Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; PK model built with macros
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
absorption(cmt=1, ka, Tlag)
elimination(cmt=1, Cl)

EQUATION:
odeType = nonStiff

; Initial condition at baseline steady state
t_0 = 0
E_0 = Rin/kout

; Inhibition of production
inh = Cc/(Cc + IC50)
ddt_E = Rin*(1 - inh) - kout*E

OUTPUT:
output = {Cc, E}
