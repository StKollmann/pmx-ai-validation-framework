DESCRIPTION:
Simeoni tumor growth inhibition model with cell distribution delay (3 death compartments)
driven by the sum of cisplatin and pemetrexed concentrations (log-kill hypothesis).

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau}

PK:
; ---- Fixed PK parameters ----
ka_cis = 66.5421
k_cis  = 0.2868
Vd_cis = 65.1131

ka_pem = 28.6
k_pem  = 2.1328
Vd_pem = 102.7673

; ---- Cisplatin: compartment 2, administration 2 ----
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; ---- Pemetrexed: compartment 3, administration 1 ----
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; ---- Initial conditions ----
TS_0  = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

psi = 20
k1 = 1/tau

; ---- Total tumor size (define before use) ----
TotalTS = TS + TS1 + TS2 + TS3

; ---- Drug exposure ----
EXPOSURE = C_cis + C_pem

; ---- Growth (exponential -> linear switch) and log-kill ----
growth = (kge*TS) / (1 + (kge/kgl*TotalTS)^psi)^(1/psi)
kill   = kkill*EXPOSURE*TS

ddt_TS  = growth - kill
ddt_TS1 = kill - k1*TS1
ddt_TS2 = k1*(TS1 - TS2)
ddt_TS3 = k1*(TS2 - TS3)

OUTPUT:
output = {TotalTS}
table = {C_cis, C_pem, EXPOSURE}
