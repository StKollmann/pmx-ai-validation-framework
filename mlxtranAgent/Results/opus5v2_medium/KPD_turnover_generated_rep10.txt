DESCRIPTION:
K-PD model: virtual dose compartment with first-order degradation (rate k),
driving a turnover (indirect response) model with inhibition of the
production (proliferation) rate through a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; Virtual K-PD compartment receiving the doses (no volume, no absorption)
depot(target=Ak)

EQUATION:
odeType = nonStiff
t_0 = 0
Ak_0 = 0
R_0 = R0

; production and loss rates of the response at baseline
kin = R0*kdeg
kout = kdeg

; virtual K-PD "concentration" = amount in the virtual compartment
ddt_Ak = -k*Ak
Ckpd = Ak

; sigmoidal Imax inhibition of the production rate
INH = Imax*Ckpd^gamma/(IC50^gamma + Ckpd^gamma)

ddt_R = kin*(1 - INH) - kout*R

OUTPUT:
output = {R}
table = {Ckpd, INH}
