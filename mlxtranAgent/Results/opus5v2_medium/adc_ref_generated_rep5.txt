DESCRIPTION:
Antibody-drug conjugate (ADC) model.
ADC: 2-compartment IV disposition (macros).
Payload: released from ADC in central compartment via proteolytic
degradation (DAR = 4 payload molecules per ADC), 1-compartment disposition (ODE).

[LONGITUDINAL]
input = {V, k10, k12, k21, V_payload, k_payload}

PK:
; --- ADC disposition written with macros ---
compartment(cmt=1, amount=A_ADC, volume=V, concentration=Cc_ADC)
iv(cmt=1, adm=1)
peripheral(k12, k21)
elimination(cmt=1, k=k10)

EQUATION:
; --- constant drug-to-antibody ratio ---
DAR = 4

; --- payload disposition (ODE), released by proteolytic catabolism of ADC ---
A_payload_0 = 0
ddt_A_payload = DAR * k10 * A_ADC - k_payload * A_payload

Cc_payload = A_payload / V_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
