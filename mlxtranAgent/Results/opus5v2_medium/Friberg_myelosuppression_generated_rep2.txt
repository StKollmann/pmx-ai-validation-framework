[LONGITUDINAL]
input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
; doses enter a hidden depot and are absorbed into Ac at rate ka
depot(target=Ac, ka)

EQUATION:
; ---- transit rate and steady-state rates ----
ktr   = 4/MTT     ; (n+1)/MTT with n = 3 transit compartments
kprol = ktr
kcirc = ktr

; ---- initial conditions ----
t_0      = 0
Ac_0     = 0
Prol_0   = Circ0
Tr1_0    = Circ0
Tr2_0    = Circ0
Tr3_0    = Circ0
Circ_0   = Circ0

; ---- PK ----
ddt_Ac = -(Cl/V)*Ac
Cc = max(Ac/V, 0)

; ---- drug effect (Emax inhibition of proliferation) ----
Edrug = Emax*Cc/(EC50 + Cc)

; ---- homeostatic feedback ----
FB = (Circ0/max(Circ, 1e-6))^gam

; ---- PD ----
ddt_Prol = kprol*Prol*(1 - Edrug)*FB - ktr*Prol
ddt_Tr1  = ktr*Prol - ktr*Tr1
ddt_Tr2  = ktr*Tr1  - ktr*Tr2
ddt_Tr3  = ktr*Tr2  - ktr*Tr3
ddt_Circ = ktr*Tr3  - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
