DESCRIPTION:
K-PD model: virtual PK compartment with first-order degradation (rate k),
driving an indirect response (turnover) model with inhibition of the
production (proliferation) rate through a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual KPD compartment receiving the doses (no volume, no absorption)
depot(target=Ak)

EQUATION:
t_0 = 0
Ak_0 = 0
R_0 = R0

; production (proliferation) rate at baseline
kin = R0*kdeg

; virtual concentration (amount in the KPD compartment)
ddt_Ak = -k*Ak

; sigmoidal Imax inhibition of the production rate
INH = Imax*Ak^gamma/(IC50^gamma + Ak^gamma)

; turnover with inhibition of production
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
