DESCRIPTION:
ADC PK model (2-compartment, IV) with proteolytic release of cytotoxic payload,
payload described by a 1-compartment model. Constant DAR = 4.

[LONGITUDINAL]
input = {k12, k21, V, k10, V_payload, k_payload}

PK:
; --- ADC disposition written with macros ---
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc_ADC)
iv(cmt=1, adm=1)
peripheral(k12, k21)
elimination(cmt=1, k=k10)

EQUATION:
; --- Constant drug-to-antibody ratio ---
DAR = 4

; --- Proteolytic release of payload from ADC in central compartment ---
release = DAR * k10 * Ac

; --- Payload disposition (1 compartment, ODE) ---
A_payload_0 = 0
ddt_A_payload = release - k_payload * A_payload

Cc_payload = A_payload / V_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
