[LONGITUDINAL]
DESCRIPTION:
Friberg semi-mechanistic myelosuppression model.
One-compartment PK with first-order absorption and linear elimination.
PD: proliferating compartment, 3 transit compartments, circulating compartment.

input = {ka, V, Cl, Circ0, MTT, gam, Emax, EC50}

PK:
; first-order absorption handled by the depot macro
depot(target=Ac, ka)

EQUATION:
; --- initial conditions ---
t_0 = 0
Ac_0 = 0
Prol_0 = Circ0
Tr1_0 = Circ0
Tr2_0 = Circ0
Tr3_0 = Circ0
Circ_0 = Circ0

; --- PK ---
Cc = Ac/V
ddt_Ac = -(Cl/V)*Ac

; --- PD intermediate variables (defined before use) ---
ktr = 4/MTT
kprol = ktr
kcirc = ktr
Edrug = Emax*Cc/(EC50 + Cc)
FB = (Circ0/max(Circ,1e-6))^gam

; --- PD system ---
ddt_Prol = kprol*Prol*(1 - Edrug)*FB - ktr*Prol
ddt_Tr1  = ktr*Prol - ktr*Tr1
ddt_Tr2  = ktr*Tr1  - ktr*Tr2
ddt_Tr3  = ktr*Tr2  - ktr*Tr3
ddt_Circ = ktr*Tr3  - kcirc*Circ

OUTPUT:
output = {Cc, Circ}
