DESCRIPTION:
Parent drug (2 compartments) with 3 metabolites (1 compartment each).
Oral dose split into fixed fractions (first-pass effect): F1 to parent,
F2, F3, F4 directly to metabolite depots. Sequential metabolism from
parent central compartment to each metabolite (kpm2, kpm3, kpm4).
Metabolite volumes fixed to parent central volume V1.

[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; fixed dose fractions (first-pass effect)
F1 = 0.1284
F2 = 0.5952
F3 = 0.2381
F4 = 0.0383

depot(target=Ad1, adm=1, p=F1)
depot(target=Ad2, adm=1, p=F2)
depot(target=Ad3, adm=1, p=F3)
depot(target=Ad4, adm=1, p=F4)

EQUATION:
odeType = stiff

; initial conditions
t_0 = 0
Ad1_0 = 0
Ad2_0 = 0
Ad3_0 = 0
Ad4_0 = 0
Ac_0  = 0
Ap_0  = 0
Am2_0 = 0
Am3_0 = 0
Am4_0 = 0

; elimination / distribution rate constants
k    = Cl/V1
k12  = Q/V1
k21  = Q/V2
km2  = Clm2/V1
km3  = Clm3/V1
km4  = Clm4/V1

; absorption compartments
ddt_Ad1 = -ka1*Ad1
ddt_Ad2 = -ka2*Ad2
ddt_Ad3 = -ka3*Ad3
ddt_Ad4 = -ka4*Ad4

; parent: central and peripheral
ddt_Ac = ka1*Ad1 - k*Ac - k12*Ac + k21*Ap - (kpm2 + kpm3 + kpm4)*Ac
ddt_Ap = k12*Ac - k21*Ap

; metabolites
ddt_Am2 = ka2*Ad2 + kpm2*Ac - km2*Am2
ddt_Am3 = ka3*Ad3 + kpm3*Ac - km3*Am3
ddt_Am4 = ka4*Ad4 + kpm4*Ac - km4*Am4

; concentrations (metabolite volumes fixed to V1)
Cp  = Ac/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
