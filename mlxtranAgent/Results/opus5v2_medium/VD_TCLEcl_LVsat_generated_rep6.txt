DESCRIPTION:
Target cell limited viral dynamics model with an eclipse phase.
Loss of virions due to infection (absorption of virus into target cells) is included.

[LONGITUDINAL]
input = {R0, kI, delta, p, c, V0, T0}

EQUATION:
odeType = stiff

; infection rate constant derived from the basic reproduction number
beta = R0*delta*c/(p*T0)

; initial conditions
t_0  = 0
T_0  = T0
I1_0 = 0
I2_0 = 0
V_0  = 10^V0

; target cells
ddt_T  = -beta*T*V
; eclipse (infected, not yet productive) cells
ddt_I1 = beta*T*V - kI*I1
; productively infected cells
ddt_I2 = kI*I1 - delta*I2
; free virus, with loss of virions due to infection
ddt_V  = p*I2 - c*V - beta*T*V

; log10 viral load with lower saturation
LV = max(log10(max(V, 1e-10)), -5)

OUTPUT:
output = {LV}
table  = {T, I1, I2, V}
