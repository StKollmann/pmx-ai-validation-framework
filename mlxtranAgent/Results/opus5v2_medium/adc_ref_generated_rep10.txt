DESCRIPTION: ADC model with 2-cmt ADC disposition and 1-cmt payload

[LONGITUDINAL]
input = {k12, k21, V, k10, V_payload, k_payload}

PK:
; Drug-to-antibody ratio (constant)
DAR = 4

; ADC disposition: central + peripheral, IV administration, linear elimination
compartment(cmt=1, amount=Ac_ADC, volume=V, concentration=Cc_ADC)
iv(cmt=1, adm=1)
peripheral(k12, k21)
elimination(cmt=1, k=k10)

EQUATION:
; Payload released from ADC in central compartment (proteolytic pathway only)
t_0 = 0
A_payload_0 = 0
ddt_A_payload = DAR*k10*Ac_ADC - k_payload*A_payload
Cc_payload = A_payload/V_payload

OUTPUT:
output = {Cc_ADC, Cc_payload}
