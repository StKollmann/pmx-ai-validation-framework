DESCRIPTION: PKPD model - 1cmt PK (1st-order absorption with lag, linear elimination) with indirect response model (inhibition of production)

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}
Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; 1-compartment model with first-order absorption and lag time
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
absorption(cmt=1, adm=1, ka, Tlag)
elimination(cmt=1, Cl)

EQUATION:
t_0 = 0
E_0 = Rin/kout

; Inhibition of production
inh = 1 - Cc/(Cc + IC50)
ddt_E = Rin*inh - kout*E

OUTPUT:
output = {Cc, E}
