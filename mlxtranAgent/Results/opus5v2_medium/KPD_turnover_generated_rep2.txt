DESCRIPTION: K-PD model: virtual PK compartment with degradation rate only (no volume,
no absorption) driving a turnover (indirect response) model with sigmoidal Imax
inhibition of the production (proliferation) rate.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual dose compartment: doses enter A directly, no volume, no absorption
depot(target=A)

EQUATION:
t_0 = 0
A_0 = 0
R_0 = R0

; baseline production rate (steady state: kin = kdeg*R0)
kin = R0*kdeg

; sigmoidal Imax inhibition driven by the virtual (K-PD) amount A
inh = Imax*A^gamma/(A^gamma + IC50^gamma)

ddt_A = -k*A
ddt_R = kin*(1 - inh) - kdeg*R

OUTPUT:
output = {R}
