DESCRIPTION:
Parent drug (2-compartment) with 3 metabolites (1 compartment each).
Oral dose is split by fixed first-pass fractions into the parent depot
and into three metabolite depots. Systemic conversion parent -> metabolites
via first-order rate constants kpm2, kpm3, kpm4.
Metabolite central volumes are fixed to the parent central volume V1.

[LONGITUDINAL]
input = {V1, Cl, Q, V2, ka1, ka2, ka3, ka4, Clm2, Clm3, Clm4, kpm2, kpm3, kpm4}

PK:
; fixed first-pass fractions
depot(target=Ad1, p=0.1284)
depot(target=Ad2, p=0.5952)
depot(target=Ad3, p=0.2381)
depot(target=Ad4, p=0.0383)

EQUATION:
odeType=stiff

; initial conditions
t_0  = 0
Ad1_0 = 0
Ac_0  = 0
Ap_0  = 0
Ad2_0 = 0
Am2_0 = 0
Ad3_0 = 0
Am3_0 = 0
Ad4_0 = 0
Am4_0 = 0

; parent
ddt_Ad1 = -ka1*Ad1
ddt_Ac  = ka1*Ad1 - (Cl/V1)*Ac - (Q/V1)*Ac + (Q/V2)*Ap - (kpm2 + kpm3 + kpm4)*Ac
ddt_Ap  = (Q/V1)*Ac - (Q/V2)*Ap

; metabolite 2
ddt_Ad2 = -ka2*Ad2
ddt_Am2 = ka2*Ad2 + kpm2*Ac - (Clm2/V1)*Am2

; metabolite 3
ddt_Ad3 = -ka3*Ad3
ddt_Am3 = ka3*Ad3 + kpm3*Ac - (Clm3/V1)*Am3

; metabolite 4
ddt_Ad4 = -ka4*Ad4
ddt_Am4 = ka4*Ad4 + kpm4*Ac - (Clm4/V1)*Am4

; concentrations
Cp  = Ac/V1
Cm2 = Am2/V1
Cm3 = Am3/V1
Cm4 = Am4/V1

OUTPUT:
output = {Cp, Cm2, Cm3, Cm4}
