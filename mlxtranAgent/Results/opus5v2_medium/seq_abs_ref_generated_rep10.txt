DESCRIPTION:
Two-compartment model with sequential zero-order then first-order absorption.
Built with PK macros only (no ODEs).

[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; inter-compartmental rate constants
k23 = Q/V
k32 = Q/V2

; absorption (depot) compartment: zero-order input, then first-order transfer
compartment(cmt=1, amount=Ad)
absorption(cmt=1, adm=1, Tk0)
transfer(from=1, to=2, kt=ka)

; central compartment
compartment(cmt=2, amount=Ac, volume=V, concentration=Cc)
elimination(cmt=2, Cl)

; peripheral compartment
peripheral(k23, k32)

OUTPUT:
output = {Cc}
