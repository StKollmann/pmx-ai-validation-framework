DESCRIPTION: One-compartment model, IV bolus, parallel linear and Michaelis-Menten elimination (ODE, amounts)

[LONGITUDINAL]
input = {V, k, Vm, Km}

PK:
; IV bolus doses go directly into the central compartment amount Ac
depot(target=Ac)

EQUATION:
t_0 = 0
Ac_0 = 0

; Define concentration BEFORE use
Cc = Ac/V

; Central compartment: linear + Michaelis-Menten elimination (amounts)
ddt_Ac = -k*Ac - Vm*Cc/(Km + Cc)

OUTPUT:
output = {Cc}
