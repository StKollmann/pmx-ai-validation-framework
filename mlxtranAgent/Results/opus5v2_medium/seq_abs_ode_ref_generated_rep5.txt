DESCRIPTION:
Two-compartment PK model with sequential zero-order then first-order absorption.
Doses enter the absorption compartment (Ad) via a zero-order process of duration Tk0,
then are absorbed into the central compartment with first-order rate constant ka.

[LONGITUDINAL]
input = {Tk0, ka, V, Cl, Q, V2}

PK:
; zero-order input of the dose into the absorption compartment Ad
depot(target=Ad, Tk0)

EQUATION:
odeType = nonStiff

; initial conditions
t_0  = 0
Ad_0 = 0
Ac_0 = 0
Ap_0 = 0

; elementary rate constants
k   = Cl/V
k12 = Q/V
k21 = Q/V2

; ODE system
ddt_Ad = -ka*Ad
ddt_Ac =  ka*Ad - k*Ac - k12*Ac + k21*Ap
ddt_Ap =  k12*Ac - k21*Ap

; central concentration
Cc = Ac/V

OUTPUT:
output = {Cc}
