DESCRIPTION:
K-PD model: virtual dose compartment with first-order degradation (k),
driving an indirect response (turnover) model with inhibition of the
production rate through a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; virtual K-PD compartment receiving the doses
depot(target=A)

EQUATION:
t_0 = 0
A_0 = 0
R_0 = R0

; virtual "concentration"/signal decaying with rate k
ddt_A = -k*A

; sigmoidal Imax inhibition of the production rate
kin = R0*kdeg
INH = Imax*max(A,0)^gamma/(IC50^gamma + max(A,0)^gamma)
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
