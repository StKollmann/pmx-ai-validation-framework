DESCRIPTION:
PKPD model: 1-compartment PK with first-order absorption and lag time, linear
elimination (PK macros, PK parameters as regressors), and an indirect response
(turnover) model with inhibition of production.

[LONGITUDINAL]
input = {Tlag, ka, V, Cl, IC50, Rin, kout}

Tlag = {use=regressor}
ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
compartment(cmt=1, amount=Ac, volume=V, concentration=Cc)
absorption(cmt=1, ka, Tlag)
elimination(cmt=1, Cl)

EQUATION:
; Baseline of the PD response
E_0 = Rin/kout

; Inhibition of the production rate
inh = 1 - Cc/(Cc + IC50)

ddt_E = Rin*inh - kout*E

OUTPUT:
output = {Cc, E}
