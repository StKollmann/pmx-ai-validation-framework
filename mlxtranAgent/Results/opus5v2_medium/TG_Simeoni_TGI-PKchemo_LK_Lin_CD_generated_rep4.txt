DESCRIPTION:
Simeoni tumor growth inhibition model driven by combined exposure of cisplatin and pemetrexed.

[LONGITUDINAL]
input = {TS0, kge, kgl, kkill, tau, ka_cis, k_cis, Vd_cis, ka_pem, k_pem, Vd_pem}

PK:
; Cisplatin PK (compartment 2, administration 2)
compartment(cmt=2, amount=A_cis, volume=Vd_cis, concentration=C_cis)
absorption(cmt=2, adm=2, ka=ka_cis)
elimination(cmt=2, k=k_cis)

; Pemetrexed PK (compartment 3, administration 1)
compartment(cmt=3, amount=A_pem, volume=Vd_pem, concentration=C_pem)
absorption(cmt=3, adm=1, ka=ka_pem)
elimination(cmt=3, k=k_pem)

EQUATION:
odeType = stiff

; initial conditions
TS_0 = TS0
TS1_0 = 0
TS2_0 = 0
TS3_0 = 0

; total tumor size (proliferating + damaged cells)
TotalTS = TS + TS1 + TS2 + TS3

; combined exposure
EXPOSURE = C_cis + C_pem

; Simeoni growth function (psi fixed to 20)
psi = 20
growth = 2*kge*TS / (1 + (2*kge*TotalTS/kgl)^psi)^(1/psi)

; log-kill effect
kill = kkill*EXPOSURE*TS

ddt_TS  = growth - kill
ddt_TS1 = kill - TS1/tau
ddt_TS2 = (TS1 - TS2)/tau
ddt_TS3 = (TS2 - TS3)/tau

OUTPUT:
output = {TotalTS}
