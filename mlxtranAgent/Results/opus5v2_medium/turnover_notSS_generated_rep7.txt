[LONGITUDINAL]
input = {ka, V, Cl, R0, Rss, kout, Emax, EC50}

ka = {use=regressor}
V = {use=regressor}
Cl = {use=regressor}

PK:
; One-compartment model with first-order absorption
Cc = pkmodel(ka, V, Cl)

EQUATION:
; Initial conditions (PD not at steady state at t=0)
t_0 = 0
R_0 = R0

; Turnover model with stimulation of production
kin = kout * Rss
Stim = Emax * Cc / (EC50 + Cc)
ddt_R = kin * (1 + Stim) - kout * R

OUTPUT:
output = {Cc, R}
