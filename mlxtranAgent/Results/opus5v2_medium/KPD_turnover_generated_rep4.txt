DESCRIPTION:
K-PD model: virtual PK compartment with first-order degradation (no volume,
no absorption) driving an indirect response (turnover) model with inhibition
of the production (proliferation) rate via a sigmoidal Imax function.

[LONGITUDINAL]
input = {k, R0, kdeg, IC50, Imax, gamma}

PK:
; doses enter the virtual K-PD compartment A
depot(target=A)

EQUATION:
odeType = nonStiff

; initial conditions
t_0 = 0
A_0 = 0
R_0 = R0

; turnover parameters
kin = R0*kdeg

; sigmoidal Imax inhibition of the production rate driven by the K-PD amount
INH = Imax*A^gamma/(IC50^gamma + A^gamma)

; virtual PK compartment (degradation only)
ddt_A = -k*A

; response
ddt_R = kin*(1 - INH) - kdeg*R

OUTPUT:
output = {R}
