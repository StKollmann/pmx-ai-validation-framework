DESCRIPTION:
One-compartment model with IV bolus administration, linear (k) and
Michaelis-Menten (Vm, Km) elimination. ODEs written in amounts.

[LONGITUDINAL]
input = {V, k, Vm, Km}

PK:
; IV bolus doses go directly into the central compartment amount Ac
depot(target=Ac)

EQUATION:
t_0 = 0
Ac_0 = 0

; Define concentration BEFORE use
Cc = Ac/V

; Elimination terms (amount per time)
elim_lin = k*Ac
elim_MM  = Vm*Cc/(Km + Cc)

ddt_Ac = -elim_lin - elim_MM

OUTPUT:
output = {Cc}
